SELECT *
from
(
with snapshots as (
select cast(20250930 as bigint) as snapshot_dt, 'train' as split
union all
select cast(20251115 as bigint) as snapshot_dt, 'validation' as split
),
base as (
select
b0.msisdn,
s.snapshot_dt,
s.split
from (
select distinct msisdn
from devdata.account_holder_dump
where tbl_dt = 20251115
and account_type = 'MOBILE MONEY'
and lower(profile) like '%agent%'
) b0
cross join snapshots s
),
win as (
select
b.msisdn,
b.snapshot_dt,
b.split,
try_cast(date_format(date_add('month', -6, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint) as dt_m6,
try_cast(date_format(date_add('month', -3, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint) as dt_m3,
try_cast(date_format(date_add('month', -1, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint) as dt_m1,
try_cast(date_format(date_add('day', -30, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint) as dt_d30,
try_cast(date_format(date_add('day', -60, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint) as dt_d60,
try_cast(date_format(date_add('day', -90, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint) as dt_d90,
try_cast(date_format(date_add('day', -120, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint) as dt_d120,
try_cast(date_format(date_add('day', -150, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint) as dt_d150,
try_cast(date_format(date_add('day', -180, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint) as dt_d180,
try_cast(date_format(date_add('day', 30, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint) as dt_p30
from base b
),
t_pull as (
select
w.msisdn,
w.snapshot_dt,
w.split,
t.tbl_dt,
t.tranname,
t.telecomid,
t.amount,
w.dt_m6, w.dt_m3, w.dt_m1,
w.dt_d30, w.dt_d60, w.dt_d90, w.dt_d120, w.dt_d150, w.dt_d180,
w.dt_p30
from win w
left join devdata.xtrafloat_daily_trans t
on t.phonenumber = w.msisdn
and t.tbl_dt > w.dt_d180
and t.tbl_dt <= w.dt_p30
and t.tranname in ('XtraFloat Dispatch', 'XtraFloat Payment', 'XtraFloat Penalty')
),
t_hist as (
select *
from t_pull
where tbl_dt <= snapshot_dt
),
t_fut as (
select *
from t_pull
where tbl_dt > snapshot_dt
),
features as (
select
msisdn as phonenumber,
snapshot_dt,
split,
max(case
when tbl_dt > dt_m6 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Dispatch'
then tbl_dt end
) as Last_disbursement_date,
max(case
when tbl_dt > dt_m6 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Payment'
then tbl_dt end
) as Last_repayment_date,
count(distinct case
when tbl_dt > dt_m1 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Penalty'
then telecomid end
) as penalties_1M,
count(distinct case
when tbl_dt > dt_m3 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Penalty'
then telecomid end
) as penalties_3M,
count(distinct case
when tbl_dt > dt_m6 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Penalty'
then telecomid end
) as penalties_6M,

count(distinct case
when tbl_dt > dt_m1 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Dispatch'
then telecomid end
) as disbursement_vol_1M,
count(distinct case
when tbl_dt > dt_d30 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Dispatch'
then telecomid end
) as disbursement_vol_M1,
count(distinct case
when tbl_dt > dt_d60 and tbl_dt <= dt_d30 and tranname = 'XtraFloat Dispatch'
then telecomid end
) as disbursement_vol_M2,
count(distinct case
when tbl_dt > dt_d90 and tbl_dt <= dt_d60 and tranname = 'XtraFloat Dispatch'
then telecomid end
) as disbursement_vol_M3,
count(distinct case
when tbl_dt > dt_d120 and tbl_dt <= dt_d90 and tranname = 'XtraFloat Dispatch'
then telecomid end
) as disbursement_vol_M4,
count(distinct case
when tbl_dt > dt_d150 and tbl_dt <= dt_d120 and tranname = 'XtraFloat Dispatch'
then telecomid end
) as disbursement_vol_M5,
count(distinct case
when tbl_dt > dt_d180 and tbl_dt <= dt_d150 and tranname = 'XtraFloat Dispatch'
then telecomid end
) as disbursement_vol_M6,
count(distinct case
when tbl_dt > dt_m3 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Dispatch'
then telecomid end
) as disbursement_vol_3M,
count(distinct case
when tbl_dt > dt_m6 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Dispatch'
then telecomid end
) as disbursement_vol_6M,
coalesce(sum(case
when tbl_dt > dt_m1 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Dispatch'
then try_cast(amount as bigint) end
), 0) as disbursement_val_1M,
coalesce(sum(case
when tbl_dt > dt_d30 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Dispatch'
then try_cast(amount as bigint) end
), 0) as disbursement_val_M1,
coalesce(sum(case
when tbl_dt > dt_d60 and tbl_dt <= dt_d30 and tranname = 'XtraFloat Dispatch'
then try_cast(amount as bigint) end
), 0) as disbursement_val_M2,
coalesce(sum(case
when tbl_dt > dt_d90 and tbl_dt <= dt_d60 and tranname = 'XtraFloat Dispatch'
then try_cast(amount as bigint) end
), 0) as disbursement_val_M3,
coalesce(sum(case
when tbl_dt > dt_d120 and tbl_dt <= dt_d90 and tranname = 'XtraFloat Dispatch'
then try_cast(amount as bigint) end
), 0) as disbursement_val_M4,
coalesce(sum(case
when tbl_dt > dt_d150 and tbl_dt <= dt_d120 and tranname = 'XtraFloat Dispatch'
then try_cast(amount as bigint) end
), 0) as disbursement_val_M5,
coalesce(sum(case
when tbl_dt > dt_d180 and tbl_dt <= dt_d150 and tranname = 'XtraFloat Dispatch'
then try_cast(amount as bigint) end
), 0) as disbursement_val_M6,
coalesce(sum(case
when tbl_dt > dt_m3 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Dispatch'
then try_cast(amount as bigint) end
), 0) as disbursement_val_3M,
coalesce(sum(case
when tbl_dt > dt_m6 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Dispatch'
then try_cast(amount as bigint) end
), 0) as disbursement_val_6M,
count(distinct case
when tbl_dt > dt_m1 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Payment'
then telecomid end
) as repayment_vol_1M,
count(distinct case
when tbl_dt > dt_d30 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Payment'
then telecomid end
) as Repayment_vol_M1,

count(distinct case
when tbl_dt > dt_d60 and tbl_dt <= dt_d30 and tranname = 'XtraFloat Payment'
then telecomid end
) as Repayment_vol_M2,
count(distinct case
when tbl_dt > dt_d90 and tbl_dt <= dt_d60 and tranname = 'XtraFloat Payment'
then telecomid end
) as Repayment_vol_M3,
count(distinct case
when tbl_dt > dt_d120 and tbl_dt <= dt_d90 and tranname = 'XtraFloat Payment'
then telecomid end
) as Repayment_vol_M4,
count(distinct case
when tbl_dt > dt_d150 and tbl_dt <= dt_d120 and tranname = 'XtraFloat Payment'
then telecomid end
) as Repayment_vol_M5,
count(distinct case
when tbl_dt > dt_d180 and tbl_dt <= dt_d150 and tranname = 'XtraFloat Payment'
then telecomid end
) as Repayment_vol_M6,
count(distinct case
when tbl_dt > dt_m3 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Payment'
then telecomid end
) as repayment_vol_3M,
count(distinct case
when tbl_dt > dt_m6 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Payment'
then telecomid end
) as repayment_vol_6M,
coalesce(sum(case
when tbl_dt > dt_m1 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Payment'
then -1 * cast(amount as double) end
), 0) as repayment_val_1M,
coalesce(sum(case
when tbl_dt > dt_d30 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Payment'
then -1 * cast(amount as double) end
), 0) as Repayment_val_M1,
coalesce(sum(case
when tbl_dt > dt_d60 and tbl_dt <= dt_d30 and tranname = 'XtraFloat Payment'
then -1 * cast(amount as double) end
), 0) as Repayment_val_M2,
coalesce(sum(case
when tbl_dt > dt_d90 and tbl_dt <= dt_d60 and tranname = 'XtraFloat Payment'
then -1 * cast(amount as double) end
), 0) as Repayment_val_M3,
coalesce(sum(case
when tbl_dt > dt_d120 and tbl_dt <= dt_d90 and tranname = 'XtraFloat Payment'
then -1 * cast(amount as double) end
), 0) as Repayment_val_M4,
coalesce(sum(case
when tbl_dt > dt_d150 and tbl_dt <= dt_d120 and tranname = 'XtraFloat Payment'
then -1 * cast(amount as double) end
), 0) as Repayment_val_M5,
coalesce(sum(case
when tbl_dt > dt_d180 and tbl_dt <= dt_d150 and tranname = 'XtraFloat Payment'
then -1 * cast(amount as double) end
), 0) as Repayment_val_M6,
coalesce(sum(case
when tbl_dt > dt_m3 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Payment'
then -1 * cast(amount as double) end
), 0) as repayment_val_3M,
coalesce(sum(case
when tbl_dt > dt_m6 and tbl_dt <= snapshot_dt and tranname = 'XtraFloat Payment'
then -1 * cast(amount as double) end
), 0) as repayment_val_6M
from t_hist
group by msisdn, snapshot_dt, split
),
/* Forward-looking label + observability flag (ONLY from t_fut, but anchored on win) */
labels as (
select
w.msisdn as phonenumber,
w.snapshot_dt,
count(distinct case
when tf.tbl_dt <= w.dt_p30 and tf.tranname = 'XtraFloat Penalty'
then tf.telecomid end
) as future_penalties_30D,
case when count(tf.tbl_dt) > 0 then 1 else 0 end as outcome_observed_30D
from win w
left join t_fut tf
on tf.msisdn = w.msisdn
and tf.snapshot_dt = w.snapshot_dt
group by w.msisdn, w.snapshot_dt
)
select distinct
b.*,
f.Last_disbursement_date,
f.disbursement_vol_1M,
f.disbursement_val_1M,
f.Last_repayment_date,
f.repayment_vol_1M,
f.repayment_val_1M,
f.penalties_1M,
f.disbursement_vol_3M,
f.disbursement_val_3M,
f.repayment_vol_3M,
f.repayment_val_3M,
f.penalties_3M,
f.disbursement_vol_6M,
f.disbursement_val_6M,
f.repayment_vol_6M,
f.repayment_val_6M,
f.penalties_6M,
f.disbursement_vol_M1,
f.disbursement_val_M1,
f.disbursement_vol_M2,
f.disbursement_val_M2,
f.disbursement_vol_M3,
f.disbursement_val_M3,
f.disbursement_vol_M4,
f.disbursement_val_M4,
f.disbursement_vol_M5,
f.disbursement_val_M5,
f.disbursement_vol_M6,
f.disbursement_val_M6,
f.Repayment_vol_M1,
f.Repayment_val_M1,
f.Repayment_vol_M2,
f.Repayment_val_M2,
f.Repayment_vol_M3,
f.Repayment_val_M3,
f.Repayment_vol_M4,
f.Repayment_val_M4,
f.Repayment_vol_M5,
f.Repayment_val_M5,
f.Repayment_vol_M6,
f.Repayment_val_M6,
l.future_penalties_30D,
case when l.future_penalties_30D > 0 then 1 else 0 end as bad_state_30D,
l.outcome_observed_30D
from base b
left join features f
on b.msisdn = f.phonenumber
and b.snapshot_dt = f.snapshot_dt
left join labels l
on b.msisdn = l.phonenumber
and b.snapshot_dt = l.snapshot_dt
where l.outcome_observed_30D = 1
) q
WHERE
CARDINALITY(
FILTER(
ARRAY[Last_disbursement_date,
disbursement_vol_1M,
disbursement_val_1M,
Last_repayment_date,
repayment_vol_1M,
repayment_val_1M,
penalties_1M,
disbursement_vol_3M,
disbursement_val_3M,
repayment_vol_3M,
repayment_val_3M,
penalties_3M,
disbursement_vol_6M,
disbursement_val_6M,
repayment_vol_6M,
repayment_val_6M,
penalties_6M,
disbursement_vol_M1,
disbursement_val_M1,
disbursement_vol_M2,
disbursement_val_M2,
disbursement_vol_M3,
disbursement_val_M3,
disbursement_vol_M4,
disbursement_val_M4,
disbursement_vol_M5,
disbursement_val_M5,
disbursement_vol_M6,
disbursement_val_M6,
Repayment_vol_M1,
Repayment_val_M1,
Repayment_vol_M2,
Repayment_val_M2,
Repayment_vol_M3,
Repayment_val_M3,
Repayment_vol_M4,
Repayment_val_M4,
Repayment_vol_M5,
Repayment_val_M5,
Repayment_vol_M6,
Repayment_val_M6],
x -> x IS NOT NULL AND x <> 0
)
) > 0;
