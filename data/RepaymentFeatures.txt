select cast(20250930 as bigint) as snapshot_dt, 'train' as split

s.snapshot_dt,
s.split
from (
select distinct msisdn
from devdata.account_holder_dump
where tbl_dt = 20251116

and try_cast(date_format(date_add('day', -30, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)

then t.telecomid end

then t.telecomid end
) as Repayment_vol_M3,
count(distinct case
when t.tbl_dt between try_cast(date_format(date_add('day', -119, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)

and t.tranname = 'XtraFloat Payment'
then t.telecomid end

when t.tbl_dt between try_cast(date_format(date_add('day', -149, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)

and t.tranname = 'XtraFloat Payment'
then t.telecomid end
) as Repayment_vol_M6,
/* repayment values */
sum(case
when t.tbl_dt between try_cast(date_format(date_add('month', -1, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and b.snapshot_dt
and t.tranname = 'XtraFloat Payment'
then -1 * cast(t.amount as double)
end) as repayment_val_1M,
sum(case
when t.tbl_dt between try_cast(date_format(date_add('day', -29, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and b.snapshot_dt
and t.tranname = 'XtraFloat Payment'
then -1 * cast(t.amount as double)
end) as Repayment_val_M1,

sum(case
when t.tbl_dt between try_cast(date_format(date_add('day', -59, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and try_cast(date_format(date_add('day', -30, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and t.tranname = 'XtraFloat Payment'
then -1 * cast(t.amount as double)
end) as Repayment_val_M2,
sum(case
when t.tbl_dt between try_cast(date_format(date_add('day', -89, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)

and try_cast(date_format(date_add('day', -90, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)

end) as Repayment_val_M4,

and try_cast(date_format(date_add('day', -120, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and t.tranname = 'XtraFloat Payment'
then -1 * cast(t.amount as double)
end) as Repayment_val_M5,
sum(case
when t.tbl_dt between try_cast(date_format(date_add('day', -179, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and try_cast(date_format(date_add('day', -150, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and t.tranname = 'XtraFloat Payment'
then -1 * cast(t.amount as double)
end) as Repayment_val_M6,
/* 3M blocks */
count(distinct case
when t.tbl_dt between try_cast(date_format(date_add('month', -3, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and b.snapshot_dt
and t.tranname = 'XtraFloat Penalty'
then t.telecomid end
) as penalties_3M,
count(distinct case
when t.tbl_dt between try_cast(date_format(date_add('month', -3, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and b.snapshot_dt
and t.tranname = 'XtraFloat Dispatch'
then t.telecomid end
) as disbursement_vol_3M,
sum(case
when t.tbl_dt between try_cast(date_format(date_add('month', -3, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and b.snapshot_dt
and t.tranname = 'XtraFloat Dispatch'
then try_cast(t.amount as bigint)
end) as disbursement_val_3M,
count(distinct case
when t.tbl_dt between try_cast(date_format(date_add('month', -3, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
and b.snapshot_dt
and t.tranname = 'XtraFloat Payment'
then t.telecomid end
) as repayment_vol_3M,

and b.snapshot_dt
and t.tranname = 'XtraFloat Payment'

end) as repayment_val_3M,
/* 6M blocks */
count(distinct case
when t.tbl_dt between try_cast(date_format(date_add('month', -6, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)

then t.telecomid end

and t.tranname = 'XtraFloat Payment'

/* forward-looking penalties in the next 30 days */

when t.tbl_dt between try_cast(date_format(date_add('day', 1, date_parse(cast(b.snapshot_dt as varchar), '%Y%m%d')), '%Y%m%d') as bigint)
