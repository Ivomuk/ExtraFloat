-- =====================================================================
-- ExtraFloat loan-level PD training dataset -- MATERIALIZED VERSION
-- SQL dialect: Trino (non-Athena; writable catalog required)
--
-- WHY THIS FILE EXISTS
--   loan_state_query_updated.txt, run as a single WITH-query, fails on
--   Trino/Athena with:
--
--     SQL Error [40]: Number of stages in the query (245) exceeds the
--     allowed maximum (100).
--
--   Trino does not materialize WITH-clause CTEs -- every reference to a
--   CTE re-expands its entire defining subquery inline. Three CTEs in
--   the original file are referenced repeatedly by downstream CTEs:
--
--     target_loans   -- 7 downstream references
--     disbursements  -- 7 downstream references
--     loan_state     -- 5 downstream references
--
--   target_loans is itself built from a ~12-CTE chain (the same-day
--   disbursement/state-loan bridge). Each of its 7 references re-runs
--   that entire chain, which is the dominant cause of the stage blow-up.
--
-- WHAT THIS FILE DOES
--   Splits the single query into five sequential statements. Run them
--   IN ORDER against the same session/connection:
--
--     1. CREATE hive.analytics.tmp_disbursements
--        (was: disbursements_ranked, disbursements)
--     2. CREATE hive.analytics.tmp_loan_state
--        (was: normalized_loan_state, loan_state_ranked, loan_state)
--     3. CREATE hive.analytics.tmp_target_loans
--        (was: the same-day bridge chain + target_loans)
--     4. CREATE hive.analytics.tmp_loan_label_assessment
--        (was: target_loan_outcomes, loan_labels_raw, loan_labels,
--        inlined in statement 4 of the earlier 4-statement version)
--        UNFILTERED -- one row per tmp_target_loans row, CENSORED_*
--        rows included, so the full population stays available for
--        audit. Eligibility filtering happens in the Python
--        preprocessing layer, not here.
--     5. Final SELECT
--        (was: everything downstream of target_loans, still is --
--        just joins tmp_loan_label_assessment for labels instead of
--        computing them inline)
--
--   Statements 4 and 5 both read from physical tables instead of
--   repeatedly-expanded CTEs, so the 7x/7x/5x duplication collapses to
--   a handful of plain table scans. Statement 4 also caps how deep the
--   label-computation CTE chain can inline ahead of statement 5, and
--   gives the label pre-flight checks (see the validation queries later
--   in this file) something physical to query standalone.
--
-- TARGET SCHEMA
--   Intermediate tables are written to hive.analytics (tmp_disbursements,
--   tmp_loan_state, tmp_target_loans, tmp_loan_label_assessment). This is
--   the same schema the source tables live in; confirm write access
--   before running, and rename these four tables if hive.analytics.tmp_*
--   could collide with anything else.
--
-- LOGIC
--   All business logic (loan matching, labels, features) is unchanged
--   from loan_state_query_updated.txt -- this file only changes *how*
--   the same computation is executed, not *what* it computes. Keep the
--   two files in sync if the logic changes.
--
-- IF STILL OVER THE STAGE LIMIT
--   The next-biggest remaining multiplier in statement 5 is
--   closed_prior_loans_ranked (3 downstream references) and prior_loans
--   (2 downstream references, feeding both prior_loans_ranked and
--   closed_prior_loans_ranked). If needed, materialize
--   hive.analytics.tmp_target_loans-derived prior_loans as its own physical
--   table the same way as the four tables below.
--
-- CLEANUP
--   hive.analytics.tmp_disbursements / tmp_loan_state / tmp_target_loans /
--   tmp_loan_label_assessment are persisted physical tables, not
--   session-scoped temp tables. Drop them when done (see the commented
--   block at the end of this file).
-- =====================================================================


-- =====================================================================
-- STATEMENT 1 of 5 -- canonical disbursement records
-- (was: disbursements_ranked, disbursements)
-- =====================================================================

DROP TABLE IF EXISTS hive.analytics.tmp_disbursements;

CREATE TABLE hive.analytics.tmp_disbursements AS
WITH disbursements_ranked AS (
    SELECT
        d.customer_msisdn AS msisdn,
        d.disbursement_fid,
        d.disbursement_uid,

        TRY_CAST(d.disbursement_amount_ugx AS DOUBLE)
            AS disbursement_amount_ugx,

        d.disbursement_ts,
        CAST(d.disbursement_ts AS DATE) AS loan_date,

        d.sales_region,
        d.sales_territory,
        d.district,
        d.inserted_ts,
        d.date_key,

        -- Policy: retain the most recently inserted physical version of
        -- each disbursement_fid. This assumes later copies are source
        -- corrections, not outcome-informed enrichment. Validate that
        -- the model attributes are invariant across duplicate physical
        -- rows.
        ROW_NUMBER() OVER (
            PARTITION BY d.disbursement_fid
            ORDER BY
                d.inserted_ts DESC,
                d.date_key DESC
        ) AS source_rn

    FROM analytics.momo_loan_book_tracker_disbursements_daily d
    WHERE d.ova = 'XTRAFLOAT-AGENT'
      AND CAST(d.disbursement_ts AS DATE) >= DATE '2026-01-01'
)

SELECT
    msisdn,
    disbursement_fid,
    disbursement_uid,
    disbursement_amount_ugx,
    disbursement_ts,
    loan_date,
    sales_region,
    sales_territory,
    district
FROM disbursements_ranked
WHERE source_rn = 1;


-- =====================================================================
-- STATEMENT 2 of 5 -- normalized, deduplicated daily state table
-- (was: normalized_loan_state, loan_state_ranked, loan_state)
-- =====================================================================

DROP TABLE IF EXISTS hive.analytics.tmp_loan_state;

CREATE TABLE hive.analytics.tmp_loan_state AS
WITH normalized_loan_state AS (
    SELECT
        ls.customer_msisdn,
        ls.loan_uid,
        ls.loan_uid_num,
        ls.loan_seq,

        ls.first_disbursement_date,
        ls.current_loan_start_date,
        ls.last_disbursement_date,
        ls.last_repayment_date,
        ls.closure_date,

        -- The source lifetime_* fields are loan-level running amounts,
        -- not aggregates across all of the customer's loans.
        TRY_CAST(ls.lifetime_disbursed_ugx AS DOUBLE)
            AS loan_disbursed_ugx,

        TRY_CAST(ls.lifetime_repaid_ugx AS DOUBLE)
            AS loan_repaid_ugx,

        TRY_CAST(ls.lifetime_gross_repaid_ugx AS DOUBLE)
            AS loan_gross_repaid_ugx,

        TRY_CAST(ls.lifetime_fees_paid_ugx AS DOUBLE)
            AS loan_fees_paid_ugx,

        TRY_CAST(ls.lifetime_interest_paid_ugx AS DOUBLE)
            AS loan_interest_paid_ugx,

        TRY_CAST(ls.outstanding_ugx AS DOUBLE)
            AS outstanding_ugx,

        ls.disbursement_event_count,
        ls.repayment_event_count,
        ls.fee_event_count,
        ls.interest_event_count,

        ls.days_aging,
        ls.aging_bucket,
        ls.loan_status,
        ls.has_pre_window_history,
        ls.is_anomaly_open,
        ls.is_active_loan,
        ls.tenure_days,

        ls.inserted_ts,

        CAST(
            DATE_PARSE(CAST(ls.date_key AS VARCHAR), '%Y%m%d')
            AS DATE
        ) AS state_date

    FROM analytics.momo_loan_book_tracker_loan_state_daily ls
    WHERE ls.ova = 'XTRAFLOAT-AGENT'
      AND ls.date_key >= 20260101
),

-- One physical state record per customer, loan_uid, and date.
loan_state_ranked AS (
    SELECT
        ls.*,

        ROW_NUMBER() OVER (
            PARTITION BY
                ls.customer_msisdn,
                ls.loan_uid,
                ls.state_date
            ORDER BY
                ls.inserted_ts DESC,
                ls.loan_seq DESC
        ) AS state_day_rn

    FROM normalized_loan_state ls
)

SELECT
    customer_msisdn,
    loan_uid,
    loan_uid_num,
    loan_seq,
    first_disbursement_date,
    current_loan_start_date,
    last_disbursement_date,
    last_repayment_date,
    closure_date,
    loan_disbursed_ugx,
    loan_repaid_ugx,
    loan_gross_repaid_ugx,
    loan_fees_paid_ugx,
    loan_interest_paid_ugx,
    outstanding_ugx,
    disbursement_event_count,
    repayment_event_count,
    fee_event_count,
    interest_event_count,
    days_aging,
    aging_bucket,
    loan_status,
    has_pre_window_history,
    is_anomaly_open,
    is_active_loan,
    tenure_days,
    inserted_ts,
    state_date
FROM loan_state_ranked
WHERE state_day_rn = 1;


-- =====================================================================
-- STATEMENT 3 of 5 -- eligible target loans, with the disbursement_fid
-- <-> loan_uid bridge resolved
-- (was: cohorts, state_data_availability, same_day_*, state_loan_*,
--       loan_id_bridge, target_loans)
--
-- CROSS-TABLE LOAN MATCHING
--   disbursement_fid is the unique loan identifier in disbursements_daily,
--   but it is not carried into loan_state_daily. loan_state_daily instead
--   generates loan_uid and supplies loan_seq.
--
--   Within each (customer, loan_date):
--     1. disbursements are ordered by disbursement_ts;
--     2. distinct state loans are ordered by loan_seq;
--     3. the Nth disbursement is mapped to the Nth state loan.
--
--   The mapping is accepted only if:
--     - every disbursement timestamp is present;
--     - disbursement timestamps are unique within the day;
--     - every state loan has a loan_seq;
--     - loan_seq is unique within the day;
--     - the number of disbursements equals the number of state loans.
--
--   Ambiguous customer-day groups are excluded rather than guessed.
-- =====================================================================

DROP TABLE IF EXISTS hive.analytics.tmp_target_loans;

-- cohort_end values below must satisfy cohort_end + 30 days <= max_state_date
-- in hive.analytics.tmp_loan_state (see the DATE_ADD guard further down and
-- state_data_availability), or that entire split comes back empty. Last
-- checked: max_state_date = 2026-06-19, so 2026-05-20 was the latest safe
-- cohort_end. Re-check max_state_date before reusing these dates later.
CREATE TABLE hive.analytics.tmp_target_loans AS
WITH cohorts AS (
    SELECT
        DATE '2026-01-01' AS cohort_start,
        DATE '2026-04-30' AS cohort_end,
        'train' AS split

    UNION ALL

    SELECT
        DATE '2026-05-01' AS cohort_start,
        DATE '2026-05-20' AS cohort_end,
        'validation' AS split
),

state_data_availability AS (
    SELECT MAX(state_date) AS max_state_date
    FROM hive.analytics.tmp_loan_state
),

-- If timestamp ties exist, positional matching is ambiguous. A stable
-- identifier is used only as a deterministic SQL tie-breaker; the group
-- will still fail the validation check and be excluded.
same_day_disbursements_ranked AS (
    SELECT
        d.*,

        ROW_NUMBER() OVER (
            PARTITION BY d.msisdn, d.loan_date
            ORDER BY
                d.disbursement_ts,
                d.disbursement_fid
        ) AS same_day_disbursement_position,

        COUNT(*) OVER (
            PARTITION BY d.msisdn, d.loan_date
        ) AS same_day_disbursement_count,

        COUNT(d.disbursement_ts) OVER (
            PARTITION BY d.msisdn, d.loan_date
        ) AS nonnull_disbursement_ts_count

    FROM hive.analytics.tmp_disbursements d
),

same_day_disbursement_validation AS (
    SELECT
        msisdn,
        loan_date,

        COUNT(*) AS disbursement_count,

        COUNT(disbursement_ts) AS nonnull_disbursement_ts_count,

        COUNT(DISTINCT disbursement_ts)
            AS distinct_disbursement_ts_count

    FROM hive.analytics.tmp_disbursements
    GROUP BY
        msisdn,
        loan_date
),

-- A loan_uid may appear on many daily rows. The identity mapping needs
-- one row per distinct state loan.
state_loan_identity_ranked AS (
    SELECT
        ls.customer_msisdn AS msisdn,
        ls.current_loan_start_date AS loan_date,
        ls.loan_uid,
        ls.loan_seq,

        ROW_NUMBER() OVER (
            PARTITION BY
                ls.customer_msisdn,
                ls.current_loan_start_date,
                ls.loan_uid
            ORDER BY
                ls.state_date,
                ls.inserted_ts
        ) AS identity_rn

    FROM hive.analytics.tmp_loan_state ls
    WHERE ls.loan_status <> 'NEVER_BORROWED'
      AND ls.current_loan_start_date IS NOT NULL
      AND ls.loan_uid IS NOT NULL
),

state_loan_identities AS (
    SELECT
        msisdn,
        loan_date,
        loan_uid,
        loan_seq
    FROM state_loan_identity_ranked
    WHERE identity_rn = 1
),

-- The mapping is rejected later if loan_seq is null or duplicated.
-- loan_uid is only a deterministic tie-breaker for SQL execution.
same_day_state_loans_ranked AS (
    SELECT
        s.msisdn,
        s.loan_date,
        s.loan_uid,
        s.loan_seq,

        ROW_NUMBER() OVER (
            PARTITION BY s.msisdn, s.loan_date
            ORDER BY
                s.loan_seq,
                s.loan_uid
        ) AS same_day_state_loan_position

    FROM state_loan_identities s
),

same_day_state_loan_validation AS (
    SELECT
        msisdn,
        loan_date,

        COUNT(*) AS state_loan_count,

        COUNT(loan_seq) AS nonnull_loan_seq_count,

        COUNT(DISTINCT loan_seq) AS distinct_loan_seq_count

    FROM state_loan_identities
    GROUP BY
        msisdn,
        loan_date
),

same_day_mapping_quality AS (
    SELECT
        d.msisdn,
        d.loan_date,
        d.disbursement_count,

        COALESCE(s.state_loan_count, 0)
            AS state_loan_count,

        CASE
            WHEN d.nonnull_disbursement_ts_count
                    = d.disbursement_count
             AND d.distinct_disbursement_ts_count
                    = d.disbursement_count

             AND COALESCE(s.nonnull_loan_seq_count, 0)
                    = COALESCE(s.state_loan_count, 0)
             AND COALESCE(s.distinct_loan_seq_count, 0)
                    = COALESCE(s.state_loan_count, 0)

             AND d.disbursement_count
                    = COALESCE(s.state_loan_count, 0)

            THEN 1
            ELSE 0
        END AS mapping_is_unambiguous

    FROM same_day_disbursement_validation d

    LEFT JOIN same_day_state_loan_validation s
      ON s.msisdn = d.msisdn
     AND s.loan_date = d.loan_date
),

loan_id_bridge AS (
    SELECT
        d.msisdn,
        d.disbursement_fid,
        d.disbursement_uid,
        d.disbursement_ts,
        d.loan_date,

        d.same_day_disbursement_position,
        d.same_day_disbursement_count,

        s.loan_uid,
        s.loan_seq AS target_loan_seq,

        q.mapping_is_unambiguous

    FROM same_day_disbursements_ranked d

    JOIN same_day_mapping_quality q
      ON q.msisdn = d.msisdn
     AND q.loan_date = d.loan_date

    LEFT JOIN same_day_state_loans_ranked s
      ON s.msisdn = d.msisdn
     AND s.loan_date = d.loan_date
     AND s.same_day_state_loan_position
            = d.same_day_disbursement_position
)

-- Ambiguous mappings are excluded and separately measurable through the
-- loan_id_bridge/mapping-quality CTEs above.
SELECT
    d.msisdn,
    d.disbursement_fid,
    d.disbursement_uid,

    b.loan_uid AS target_loan_uid,
    b.target_loan_seq,
    b.same_day_disbursement_position,
    b.same_day_disbursement_count,

    c.split,

    d.disbursement_ts,
    d.loan_date,

    DATE_ADD('day', 7, d.loan_date)
        AS label_horizon_7d_end,

    DATE_ADD('day', 30, d.loan_date)
        AS label_horizon_30d_end,

    d.disbursement_amount_ugx,
    d.sales_region,
    d.sales_territory,
    d.district

FROM hive.analytics.tmp_disbursements d

JOIN loan_id_bridge b
  ON b.disbursement_fid = d.disbursement_fid
 AND b.msisdn = d.msisdn

JOIN cohorts c
  ON d.loan_date BETWEEN c.cohort_start AND c.cohort_end

CROSS JOIN state_data_availability a

WHERE b.mapping_is_unambiguous = 1
  AND b.loan_uid IS NOT NULL
  AND DATE_ADD('day', 30, d.loan_date) <= a.max_state_date;


-- =====================================================================
-- STATEMENT 4 of 5 -- loan label assessment (outcome, coverage, and
-- eligibility for every target loan)
-- (was: target_loan_outcomes, loan_labels_raw, loan_labels, previously
--       inlined into statement 4 of the earlier 4-statement version)
--
-- UNFILTERED BY DESIGN
--   One row per hive.analytics.tmp_target_loans row -- CENSORED_* rows
--   are included, not dropped, so the full population stays available
--   for audit. Eligibility filtering happens downstream, in the Python
--   preprocessing layer (pd_model/preprocessing/loan_history_features.py),
--   not here.
--
-- PRIMARY LABEL
--   bad_state_3dpd_30d = 1 if the mapped target loan, during days 1-30:
--     - reaches days_aging > 3; or
--     - becomes ANOMALY_OPEN / is_anomaly_open.
--
-- SECONDARY LABEL
--   bad_state_1dpd_7d = 1 if the mapped target loan, during days 1-7:
--     - reaches days_aging >= 1; or
--     - becomes ANOMALY_OPEN / is_anomaly_open.
--
-- LABEL-ELIGIBILITY POLICY (asymmetric censoring, standard for
-- credit-risk problems with incomplete follow-up)
--   A "known bad" is trusted regardless of follow-up completeness --
--   bad_state_3dpd_30d is derived only from state rows strictly after
--   the loan date, so any bad signal observed at all is real evidence,
--   incomplete follow-up or not.
--
--   A "confirmed good" additionally requires:
--     - at least one post-disbursement observed date
--       (observed_date_count_to_required_end_30d > 0);
--     - a bounded follow-up coverage ratio >= 0.8
--       (meets_coverage_ratio_30d -- the ratio is defined once, in the
--       loan_label_coverage_ratio stage below, and every downstream
--       stage references that single boolean rather than re-deriving
--       the 0.8 threshold);
--     - the last observed date within 5 days of the required
--       observation end (near_horizon_observation_30d) -- otherwise
--       there is no way to distinguish "confirmed good" from "we
--       simply stopped observing this loan".
--   OR, independently of all three of the above,
--   same_day_settlement_observed_30d = 1: a terminal status
--   (SETTLED/CLOSED/OVERPAID) observed with a known, near-zero balance
--   (outstanding_ugx IS NOT NULL AND <= 0.01) on state_date = loan_date
--   itself (day 0). This is a 24-hour agent-float product, so same-day
--   settlement is the expected best outcome, not sparse follow-up --
--   without this path, observed_date_count_to_required_end_30d (gated to
--   state_date > loan_date) is structurally always 0 for such a loan,
--   since the source table stops emitting rows for a loan_uid the day
--   after it closes, and the loan could never reach confirmed_good_30d=1
--   via the normal coverage-ratio path. The NULL balance check fails
--   closed (an unknown balance is not treated as zero).
--
--   required_observation_end_date_30d shortens the required window to
--   a genuinely observed closure date when one is seen inside the
--   30-day horizon (ls.state_date >= ls.closure_date guards against
--   trusting a backfilled/provisional closure_date before it was
--   actually observed); otherwise it stays the full
--   label_horizon_30d_end. This lets a loan that closes early get a
--   correspondingly shorter, achievable observation requirement instead
--   of always being held to the full 30-day window.
--
--   Every row carries label_eligible_30d (1 iff KNOWN_BAD or
--   CONFIRMED_GOOD_*) and label_eligibility_reason_30d (one of
--   KNOWN_BAD / CONFIRMED_GOOD_SAME_DAY / CONFIRMED_GOOD_TERMINAL /
--   CONFIRMED_GOOD_HORIZON / CENSORED_NO_POST_DISBURSEMENT_STATE /
--   CENSORED_INVALID_OBSERVATION_WINDOW /
--   CENSORED_SPARSE_FOLLOW_UP / CENSORED_EARLY_LAST_OBSERVATION /
--   CENSORED_OTHER). CENSORED_INVALID_OBSERVATION_WINDOW is a genuine
--   data-integrity signal (required_observation_end_date_30d is
--   constructed to always be strictly after loan_date, so a NULL ratio
--   here means something upstream is broken) -- see the pre-flight
--   validation queries later in this file; a nonzero count there is a
--   build-blocking failure, not ordinary censoring.
--
-- ROLLOVER LIMITATION
--   The first ANOMALY_OPEN state on the mapped loan_uid is detected.
--   If the transferred debt later continues under a regenerated loan_uid,
--   subsequent aging on the successor loan is not attributed to the
--   original loan unless the source supplies a predecessor/root-loan
--   relationship. Therefore, rollover detection should remain part of
--   the binary label, while post-rollover max aging requires warehouse
--   validation.
-- =====================================================================

CREATE TABLE hive.analytics.tmp_loan_label_assessment AS
WITH target_loan_outcome_base AS (
    SELECT
        d.disbursement_fid,
        d.msisdn,
        d.split,
        d.target_loan_uid,
        d.loan_date,
        d.label_horizon_7d_end,
        d.label_horizon_30d_end,

        COUNT(DISTINCT CASE
            WHEN ls.state_date > d.loan_date
             AND ls.state_date <= d.label_horizon_7d_end
            THEN ls.state_date
        END) AS post_disbursement_observed_date_count_7d,

        COUNT(DISTINCT CASE
            WHEN ls.state_date > d.loan_date
             AND ls.state_date <= d.label_horizon_30d_end
            THEN ls.state_date
        END) AS post_disbursement_observed_date_count_30d,

        MAX(CASE
            WHEN ls.state_date > d.loan_date
             AND ls.state_date <= d.label_horizon_30d_end
            THEN ls.state_date
        END) AS last_post_disbursement_state_date_30d,

        COUNT(ls.state_date) AS outcome_state_row_count_30d,
        COUNT(DISTINCT ls.state_date) AS outcome_observed_date_count_30d,
        MIN(ls.state_date) AS first_outcome_state_date,
        MAX(ls.state_date) AS last_outcome_state_date,

        MAX(CASE
            WHEN ls.state_date > d.loan_date
             AND ls.state_date <= d.label_horizon_7d_end
            THEN ls.days_aging
        END) AS max_days_aging_7d,

        MAX(CASE
            WHEN ls.state_date > d.loan_date
             AND ls.state_date <= d.label_horizon_30d_end
            THEN ls.days_aging
        END) AS max_days_aging_30d,

        MAX(CASE
            WHEN ls.state_date > d.loan_date
             AND ls.state_date <= d.label_horizon_7d_end
             AND (
                    ls.loan_status = 'ANOMALY_OPEN'
                    OR COALESCE(ls.is_anomaly_open, FALSE)
                 )
            THEN 1
            ELSE 0
        END) AS rollover_observed_7d,

        MAX(CASE
            WHEN ls.state_date > d.loan_date
             AND ls.state_date <= d.label_horizon_30d_end
             AND (
                    ls.loan_status = 'ANOMALY_OPEN'
                    OR COALESCE(ls.is_anomaly_open, FALSE)
                 )
            THEN 1
            ELSE 0
        END) AS rollover_observed_30d,

        -- loan_status IN (...) is the sole acceptance path -- a
        -- standalone closure_date-populated OR-branch existed here
        -- previously but was removed after a direct data check confirmed
        -- the source table populates closure_date on genuinely
        -- non-terminal rows too (observed on ANOMALY_OPEN rows, holding a
        -- stale/unrelated date while the loan was still open) --
        -- closure_date alone is not trustworthy evidence of a terminal
        -- state without loan_status confirming it.
        MAX(CASE
            WHEN ls.state_date > d.loan_date
             AND ls.state_date <= d.label_horizon_30d_end
             AND ls.loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
            THEN 1
            ELSE 0
        END) AS terminal_state_observed_30d,

        -- Day-0-only evidence for a same-day-settled loan. This is a
        -- 24-hour agent-float product -- same-day settlement is the
        -- expected best outcome, not sparse follow-up. Deliberately keyed
        -- off ls.state_date = d.loan_date (day 0 itself), the one date
        -- every other post-disbursement aggregate in this CTE excludes by
        -- design. Without this flag, observed_date_count_to_required_end_30d
        -- (gated to state_date > loan_date) is structurally always 0 for
        -- such a loan, since the upstream source table stops emitting
        -- rows for a loan_uid the day after it closes -- confirmed_good_30d
        -- could then never be reached via the normal coverage-ratio path.
        -- The outstanding_ugx checks are defense-in-depth against a
        -- status/balance inconsistency: IS NOT NULL is required (not
        -- COALESCE'd to 0) because outstanding_ugx is TRY_CAST'd in
        -- statement 2, so a malformed source value silently becomes NULL
        -- there -- COALESCE-ing an unknown balance to 0 here would treat
        -- "we don't know the balance" as "fully repaid" and fail open,
        -- exactly backwards from the intent. A negative balance (OVERPAID)
        -- still passes <= 0.01. If either check fails, this flag stays 0
        -- and the loan falls through to the existing, unaffected
        -- CENSORED_NO_POST_DISBURSEMENT_STATE bucket rather than being
        -- misclassified as confirmed-good.
        MAX(CASE
            WHEN ls.state_date = d.loan_date
             AND ls.loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
             AND ls.outstanding_ugx IS NOT NULL
             AND ls.outstanding_ugx <= 0.01
            THEN 1
            ELSE 0
        END) AS same_day_settlement_observed_30d,

        -- Genuinely observed closure inside the 30-day horizon.
        -- ls.loan_status IN (...) is required, not merely ls.closure_date
        -- being populated -- a direct data check confirmed the source
        -- table sets closure_date on genuinely non-terminal rows too
        -- (e.g. ANOMALY_OPEN rows carrying a stale/unrelated date while
        -- the loan was still open), so closure_date alone is not
        -- reliable evidence the loan actually closed on that date.
        -- state_date >= closure_date is additional defense-in-depth
        -- against trusting a provisional/backfilled closure_date before
        -- it was actually recorded as observed, even on a genuinely
        -- terminal row.
        MIN(CASE
            WHEN ls.state_date > d.loan_date
             AND ls.state_date <= d.label_horizon_30d_end
             AND ls.loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
             AND ls.closure_date > d.loan_date
             AND ls.closure_date <= d.label_horizon_30d_end
             AND ls.state_date >= ls.closure_date
            THEN ls.closure_date
        END) AS observed_closure_date_30d,

        MIN(CASE
            WHEN ls.state_date > d.loan_date
             AND ls.state_date <= d.label_horizon_30d_end
             AND ls.loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
             AND ls.closure_date > d.loan_date
             AND ls.closure_date <= d.label_horizon_30d_end
             AND ls.state_date >= ls.closure_date
            THEN ls.state_date
        END) AS closure_observation_state_date_30d

    FROM hive.analytics.tmp_target_loans d

    LEFT JOIN hive.analytics.tmp_loan_state ls
      ON ls.customer_msisdn = d.msisdn
     AND ls.loan_uid = d.target_loan_uid
     AND ls.state_date >= d.loan_date
     AND ls.state_date <= d.label_horizon_30d_end

    GROUP BY
        d.disbursement_fid,
        d.msisdn,
        d.split,
        d.target_loan_uid,
        d.loan_date,
        d.label_horizon_7d_end,
        d.label_horizon_30d_end
),

-- required_observation_end_date_30d: the full 30-day horizon, shortened
-- to a genuinely observed closure date when one occurred inside it.
target_loan_outcome_window AS (
    SELECT b.*,
        COALESCE(b.observed_closure_date_30d, b.label_horizon_30d_end)
            AS required_observation_end_date_30d
    FROM target_loan_outcome_base b
),

-- Observation coverage up to the (possibly shortened) required end.
-- Explicit column list, not w.*, so a future addition to
-- target_loan_outcome_base can't silently desync from this GROUP BY.
target_loan_coverage AS (
    SELECT
        w.disbursement_fid, w.msisdn, w.split, w.target_loan_uid, w.loan_date,
        w.label_horizon_7d_end, w.label_horizon_30d_end,
        w.max_days_aging_7d, w.max_days_aging_30d,
        w.rollover_observed_7d, w.rollover_observed_30d,
        w.terminal_state_observed_30d, w.same_day_settlement_observed_30d,
        w.outcome_state_row_count_30d, w.outcome_observed_date_count_30d,
        w.post_disbursement_observed_date_count_7d, w.post_disbursement_observed_date_count_30d,
        w.last_post_disbursement_state_date_30d,
        w.first_outcome_state_date, w.last_outcome_state_date,
        w.observed_closure_date_30d, w.closure_observation_state_date_30d,
        w.required_observation_end_date_30d,

        COUNT(DISTINCT CASE
            WHEN ls.state_date > w.loan_date
            THEN ls.state_date
        END) AS observed_date_count_to_required_end_30d,

        MAX(CASE
            WHEN ls.state_date > w.loan_date
            THEN ls.state_date
        END) AS last_state_date_to_required_end_30d

    FROM target_loan_outcome_window w

    LEFT JOIN hive.analytics.tmp_loan_state ls
      ON ls.customer_msisdn = w.msisdn
     AND ls.loan_uid = w.target_loan_uid
     AND ls.state_date >= w.loan_date
     AND ls.state_date <= w.required_observation_end_date_30d

    GROUP BY
        w.disbursement_fid, w.msisdn, w.split, w.target_loan_uid, w.loan_date,
        w.label_horizon_7d_end, w.label_horizon_30d_end,
        w.max_days_aging_7d, w.max_days_aging_30d,
        w.rollover_observed_7d, w.rollover_observed_30d,
        w.terminal_state_observed_30d, w.same_day_settlement_observed_30d,
        w.outcome_state_row_count_30d, w.outcome_observed_date_count_30d,
        w.post_disbursement_observed_date_count_7d, w.post_disbursement_observed_date_count_30d,
        w.last_post_disbursement_state_date_30d,
        w.first_outcome_state_date, w.last_outcome_state_date,
        w.observed_closure_date_30d, w.closure_observation_state_date_30d,
        w.required_observation_end_date_30d
),

loan_labels_raw AS (
    SELECT
        c.*,

        CASE WHEN COALESCE(c.max_days_aging_7d, 0) >= 1 OR c.rollover_observed_7d = 1
             THEN 1 ELSE 0 END AS bad_state_1dpd_7d,
        CASE WHEN COALESCE(c.max_days_aging_30d, 0) > 3 OR c.rollover_observed_30d = 1
             THEN 1 ELSE 0 END AS bad_state_3dpd_30d,

        DATE_DIFF('day', c.loan_date, c.required_observation_end_date_30d)
            AS expected_observed_date_count_30d,

        -- Maximum allowed endpoint lag of 5 days (inclusive bound
        -- accepts 6 trailing dates, not literally "within 5 days").
        CASE WHEN c.observed_date_count_to_required_end_30d > 0
              AND c.last_state_date_to_required_end_30d >= GREATEST(
                    DATE_ADD('day', 1, c.loan_date),
                    DATE_ADD('day', -5, c.required_observation_end_date_30d)
                  )
             THEN 1 ELSE 0 END AS near_horizon_observation_30d

    FROM target_loan_coverage c
),

loan_label_coverage_ratio AS (
    SELECT r.*,
        TRY_CAST(r.observed_date_count_to_required_end_30d AS DOUBLE)
            / NULLIF(r.expected_observed_date_count_30d, 0)
            AS follow_up_coverage_ratio_30d
    FROM loan_labels_raw r
),

-- The ONLY place 0.8 appears. Every downstream stage references
-- meets_coverage_ratio_30d, never the literal.
loan_label_coverage_assessment AS (
    SELECT r.*,
        CASE WHEN r.follow_up_coverage_ratio_30d >= 0.8
             THEN 1 ELSE 0 END AS meets_coverage_ratio_30d
    FROM loan_label_coverage_ratio r
),

loan_label_confirmation AS (
    SELECT g.*,
        CASE WHEN g.bad_state_3dpd_30d = 0
              AND (
                    (
                        g.observed_date_count_to_required_end_30d > 0
                    AND g.meets_coverage_ratio_30d = 1
                    AND g.near_horizon_observation_30d = 1
                    )
                 OR g.same_day_settlement_observed_30d = 1
                 )
             THEN 1 ELSE 0 END AS confirmed_good_30d
    FROM loan_label_coverage_assessment g
),

-- Unfiltered -- every target loan, CENSORED_* rows included. This
-- becomes hive.analytics.tmp_loan_label_assessment in full.
loan_label_assessment AS (
    SELECT h.*,
        CASE WHEN h.observed_date_count_to_required_end_30d = 0 THEN 0 ELSE 1 END
            AS has_post_disbursement_state_30d,

        CASE WHEN h.meets_coverage_ratio_30d = 1 THEN 0 ELSE 1 END
            AS fails_coverage_ratio_30d,

        CASE WHEN h.near_horizon_observation_30d = 0 THEN 1 ELSE 0 END
            AS fails_near_horizon_30d,

        CASE WHEN h.bad_state_3dpd_30d = 1 OR h.confirmed_good_30d = 1
             THEN 1 ELSE 0 END AS label_eligible_30d,

        CASE
            WHEN h.bad_state_3dpd_30d = 1 THEN 'KNOWN_BAD'
            -- Must precede CONFIRMED_GOOD_HORIZON: once
            -- same_day_settlement_observed_30d is OR'd into
            -- confirmed_good_30d (stage 7), HORIZON's condition
            -- (confirmed_good_30d=1 AND closure NULL) becomes a strict
            -- superset of this branch's condition and would otherwise
            -- mask it entirely. Position relative to TERMINAL is
            -- immaterial: observed_closure_date_30d requires
            -- closure_date > loan_date, so it is always NULL when the
            -- loan's only evidence is on day 0 -- TERMINAL structurally
            -- never matches a same-day-only row.
            WHEN h.confirmed_good_30d = 1 AND h.same_day_settlement_observed_30d = 1
                THEN 'CONFIRMED_GOOD_SAME_DAY'
            WHEN h.confirmed_good_30d = 1 AND h.observed_closure_date_30d IS NOT NULL
                THEN 'CONFIRMED_GOOD_TERMINAL'
            WHEN h.confirmed_good_30d = 1 THEN 'CONFIRMED_GOOD_HORIZON'
            WHEN h.observed_date_count_to_required_end_30d = 0
                THEN 'CENSORED_NO_POST_DISBURSEMENT_STATE'
            WHEN h.follow_up_coverage_ratio_30d IS NULL
                THEN 'CENSORED_INVALID_OBSERVATION_WINDOW'
            WHEN h.meets_coverage_ratio_30d = 0
                THEN 'CENSORED_SPARSE_FOLLOW_UP'
            WHEN h.near_horizon_observation_30d = 0
                THEN 'CENSORED_EARLY_LAST_OBSERVATION'
            ELSE 'CENSORED_OTHER'
        END AS label_eligibility_reason_30d
    FROM loan_label_confirmation h
)

SELECT * FROM loan_label_assessment;

-- SCORING FEATURES

WITH

-- ---------------------------------------------------------------------
-- State known at scoring
--
-- Because state data is daily, only dates strictly before disbursement
-- are used. This prevents leakage from a same-day snapshot generated
-- after the disbursement.
-- ---------------------------------------------------------------------
scoring_state_candidates AS (
    SELECT
        d.disbursement_fid,

        ls.state_date,
        ls.loan_uid,
        ls.loan_seq,
        ls.current_loan_start_date,

        ls.loan_status,
        ls.is_active_loan,
        ls.is_anomaly_open,
        ls.has_pre_window_history,

        ls.days_aging,
        ls.aging_bucket,
        ls.outstanding_ugx,

        ls.last_repayment_date,
        ls.closure_date,
        ls.tenure_days,

        ls.loan_disbursed_ugx,
        ls.loan_repaid_ugx,
        ls.loan_gross_repaid_ugx,
        ls.loan_fees_paid_ugx,
        ls.loan_interest_paid_ugx,

        ls.disbursement_event_count,
        ls.repayment_event_count,
        ls.fee_event_count,
        ls.interest_event_count,

        ROW_NUMBER() OVER (
            PARTITION BY d.disbursement_fid
            ORDER BY
                ls.state_date DESC,
                ls.inserted_ts DESC,
                ls.loan_seq DESC,
                ls.loan_uid DESC
        ) AS scoring_state_rn

    FROM hive.analytics.tmp_target_loans d

    JOIN hive.analytics.tmp_loan_state ls
      ON ls.customer_msisdn = d.msisdn
     AND ls.state_date < d.loan_date
     AND ls.loan_status <> 'NEVER_BORROWED'
),

scoring_state AS (
    SELECT *
    FROM scoring_state_candidates
    WHERE scoring_state_rn = 1
),

-- ---------------------------------------------------------------------
-- Scoring-time features
-- ---------------------------------------------------------------------
scoring_state_features AS (
    SELECT
        d.disbursement_fid,

        CASE
            WHEN s.disbursement_fid IS NULL THEN 0
            ELSE 1
        END AS has_observed_prior_state,

        COALESCE(
            CASE WHEN s.has_pre_window_history THEN 1 ELSE 0 END,
            0
        ) AS has_pre_window_history,

        CASE
            WHEN COALESCE(s.is_active_loan, FALSE)
              OR s.loan_status IN (
                    'OPEN',
                    'OPEN_PRE_WINDOW',
                    'ANOMALY_OPEN'
                 )
            THEN 1
            ELSE 0
        END AS has_unresolved_loan_at_scoring,

        CASE
            WHEN COALESCE(s.is_active_loan, FALSE)
              OR s.loan_status IN (
                    'OPEN',
                    'OPEN_PRE_WINDOW',
                    'ANOMALY_OPEN'
                 )
            THEN COALESCE(s.days_aging, 0)
            ELSE 0
        END AS active_loan_days_aging_at_scoring,

        CASE
            WHEN COALESCE(s.is_active_loan, FALSE)
              OR s.loan_status IN (
                    'OPEN',
                    'OPEN_PRE_WINDOW',
                    'ANOMALY_OPEN'
                 )
            THEN COALESCE(s.outstanding_ugx, 0)
            ELSE 0
        END AS active_loan_outstanding_ugx_at_scoring,

        CASE
            WHEN COALESCE(s.is_active_loan, FALSE)
              OR s.loan_status IN (
                    'OPEN',
                    'OPEN_PRE_WINDOW',
                    'ANOMALY_OPEN'
                 )
            THEN s.aging_bucket
        END AS active_loan_aging_bucket_at_scoring,

        CASE
            WHEN s.loan_status = 'ANOMALY_OPEN'
              OR COALESCE(s.is_anomaly_open, FALSE)
            THEN 1
            ELSE 0
        END AS anomaly_open_at_scoring,

        s.state_date AS scoring_state_date,
        s.loan_uid AS scoring_state_loan_uid,
        s.loan_seq AS loan_seq_at_scoring,
        s.loan_status AS scoring_state_loan_status,

        CASE
            WHEN s.last_repayment_date IS NOT NULL
            THEN DATE_DIFF(
                'day',
                s.last_repayment_date,
                d.loan_date
            )
        END AS days_since_last_repayment,

        s.loan_disbursed_ugx
            AS scoring_state_loan_disbursed_ugx,

        s.loan_repaid_ugx
            AS scoring_state_loan_repaid_ugx,

        s.loan_gross_repaid_ugx
            AS scoring_state_loan_gross_repaid_ugx,

        s.loan_fees_paid_ugx
            AS scoring_state_loan_fees_paid_ugx,

        s.loan_interest_paid_ugx
            AS scoring_state_loan_interest_paid_ugx,

        CASE
            WHEN s.loan_disbursed_ugx > 0
            THEN s.loan_repaid_ugx / s.loan_disbursed_ugx
        END AS scoring_state_loan_principal_repayment_ratio,

        CASE
            WHEN s.loan_disbursed_ugx > 0
            THEN s.loan_gross_repaid_ugx / s.loan_disbursed_ugx
        END AS scoring_state_loan_gross_repayment_ratio,

        CASE
            WHEN s.disbursement_event_count > 0
            THEN
                TRY_CAST(s.repayment_event_count AS DOUBLE)
                / TRY_CAST(s.disbursement_event_count AS DOUBLE)
        END AS scoring_state_repayment_to_disbursement_event_ratio

    FROM hive.analytics.tmp_target_loans d

    LEFT JOIN scoring_state s
      ON s.disbursement_fid = d.disbursement_fid
),

-- ---------------------------------------------------------------------
-- One canonical state per prior loan
--
-- Earlier-date prior loans:
--   Use only state snapshots strictly before the target loan date.
--
-- Earlier same-day prior loans:
--   A customer may repay loan 1 and receive loan 2 later that day.
--   For loan 2, loan 1 may be included as prior history when:
--     - it has a lower loan_seq than the mapped target loan;
--     - it has closure evidence on that date; and
--     - its state row is dated no later than the target loan date.
--
-- Because loan_state_daily has dates rather than event timestamps, this
-- proves same-day closure but does not independently prove that closure
-- occurred before loan 2's exact disbursement timestamp. The positional
-- mapping and product rule supply that ordering assumption.
--
-- The target loan_uid itself is always excluded.
-- ---------------------------------------------------------------------
prior_loan_state_candidates AS (
    SELECT
        d.disbursement_fid AS target_disbursement_fid,

        ls.loan_uid,
        ls.loan_seq,
        ls.current_loan_start_date,
        ls.closure_date,
        ls.tenure_days,
        ls.aging_bucket,
        ls.loan_status,
        ls.is_anomaly_open,
        ls.days_aging,
        ls.outstanding_ugx,

        ls.loan_disbursed_ugx,
        ls.loan_repaid_ugx,
        ls.loan_gross_repaid_ugx,
        ls.loan_fees_paid_ugx,
        ls.loan_interest_paid_ugx,

        ROW_NUMBER() OVER (
            PARTITION BY
                d.disbursement_fid,
                ls.loan_uid
            ORDER BY
                ls.state_date DESC,
                ls.inserted_ts DESC
        ) AS prior_loan_state_rn

    FROM hive.analytics.tmp_target_loans d

    JOIN hive.analytics.tmp_loan_state ls
      ON ls.customer_msisdn = d.msisdn
     AND ls.loan_uid <> d.target_loan_uid
     AND ls.loan_status <> 'NEVER_BORROWED'
     AND (
            -- Loans originating before the target date use only state
            -- known strictly before that date.
            (
                ls.current_loan_start_date < d.loan_date
                AND ls.state_date < d.loan_date
            )

            OR

            -- For target loan 2 on a customer-day, permit the closed
            -- state of loan 1 from that same day.
            (
                ls.current_loan_start_date = d.loan_date
                AND ls.state_date <= d.loan_date
                AND ls.loan_seq < d.target_loan_seq
                AND ls.closure_date IS NOT NULL
                AND ls.closure_date <= d.loan_date
            )
         )
),

prior_loans AS (
    SELECT
        target_disbursement_fid,
        loan_uid,
        loan_seq,
        current_loan_start_date,
        closure_date,
        tenure_days,
        aging_bucket,
        loan_status,
        is_anomaly_open,
        days_aging,
        outstanding_ugx,

        loan_disbursed_ugx,
        loan_repaid_ugx,
        loan_gross_repaid_ugx,
        loan_fees_paid_ugx,
        loan_interest_paid_ugx

    FROM prior_loan_state_candidates
    WHERE prior_loan_state_rn = 1
),

prior_loans_ranked AS (
    SELECT
        p.*,

        ROW_NUMBER() OVER (
            PARTITION BY p.target_disbursement_fid
            ORDER BY
                p.current_loan_start_date DESC,
                p.loan_seq DESC,
                p.loan_uid DESC
        ) AS prior_loan_recency

    FROM prior_loans p
),

-- Closed loans get a separate ranking. Open loans cannot consume any of
-- the five positions used by max_tenure_days_last_5_closed_loans.
-- loan_status IN (...) is required alongside closure_date IS NOT NULL --
-- confirmed via a direct data check that the source table populates
-- closure_date on genuinely non-terminal rows too (e.g. ANOMALY_OPEN
-- rows carrying a stale/unrelated date while the loan was still open),
-- so closure_date alone is not reliable evidence a loan actually closed
-- (same finding and fix as target_loan_outcome_base's
-- observed_closure_date_30d above).
closed_prior_loans_ranked AS (
    SELECT
        p.*,

        ROW_NUMBER() OVER (
            PARTITION BY p.target_disbursement_fid
            ORDER BY
                p.closure_date DESC,
                p.current_loan_start_date DESC,
                p.loan_seq DESC,
                p.loan_uid DESC
        ) AS closed_loan_recency

    FROM prior_loans p
    WHERE p.closure_date IS NOT NULL
      AND p.loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
),

-- ---------------------------------------------------------------------
-- Features over all unique prior loans
-- ---------------------------------------------------------------------
prior_loan_features AS (
    SELECT
        target_disbursement_fid,

        COUNT(*) AS observed_prior_loan_count,

        -- loan_status IN (...) required alongside closure_date IS NOT
        -- NULL in every "closed loan" condition below -- see
        -- closed_prior_loans_ranked's comment above for why.
        COUNT_IF(
            closure_date IS NOT NULL
            AND loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
        ) AS prior_closed_loan_count,

        AVG(
            CASE
                WHEN closure_date IS NOT NULL
                 AND loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
                THEN TRY_CAST(tenure_days AS DOUBLE)
            END
        ) AS avg_tenure_days_closed_loans,

        SUM(
            CASE
                WHEN loan_status = 'ANOMALY_OPEN'
                  OR COALESCE(is_anomaly_open, FALSE)
                THEN 1
                ELSE 0
            END
        ) AS historical_anomaly_open_loan_count,

        AVG(
            CASE
                WHEN loan_status = 'ANOMALY_OPEN'
                  OR COALESCE(is_anomaly_open, FALSE)
                THEN 1.0
                ELSE 0.0
            END
        ) AS historical_anomaly_open_loan_rate,

        SUM(COALESCE(loan_disbursed_ugx, 0))
            AS all_prior_loans_disbursed_ugx,

        SUM(COALESCE(loan_repaid_ugx, 0))
            AS all_prior_loans_repaid_ugx,

        SUM(COALESCE(loan_gross_repaid_ugx, 0))
            AS all_prior_loans_gross_repaid_ugx,

        SUM(COALESCE(loan_fees_paid_ugx, 0))
            AS all_prior_loans_fees_paid_ugx,

        SUM(COALESCE(loan_interest_paid_ugx, 0))
            AS all_prior_loans_interest_paid_ugx,

        CASE
            WHEN SUM(COALESCE(loan_disbursed_ugx, 0)) > 0
            THEN
                SUM(COALESCE(loan_repaid_ugx, 0))
                / SUM(COALESCE(loan_disbursed_ugx, 0))
        END AS all_prior_loans_principal_repayment_ratio,

        CASE
            WHEN SUM(COALESCE(loan_disbursed_ugx, 0)) > 0
            THEN
                SUM(COALESCE(loan_gross_repaid_ugx, 0))
                / SUM(COALESCE(loan_disbursed_ugx, 0))
        END AS all_prior_loans_gross_repayment_ratio,

        AVG(
            CASE
                WHEN loan_disbursed_ugx > 0
                THEN loan_repaid_ugx / loan_disbursed_ugx
            END
        ) AS avg_prior_loan_principal_repayment_ratio,

        -- Closed-loan-only ratio avoids mechanically penalizing an agent
        -- merely because an unresolved loan exists at scoring.
        -- loan_status IN (...) required alongside closure_date IS NOT
        -- NULL -- see closed_prior_loans_ranked's comment above for why.
        CASE
            WHEN SUM(
                CASE
                    WHEN closure_date IS NOT NULL
                     AND loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
                    THEN COALESCE(loan_disbursed_ugx, 0)
                    ELSE 0
                END
            ) > 0
            THEN
                SUM(
                    CASE
                        WHEN closure_date IS NOT NULL
                         AND loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
                        THEN COALESCE(loan_repaid_ugx, 0)
                        ELSE 0
                    END
                )
                /
                SUM(
                    CASE
                        WHEN closure_date IS NOT NULL
                         AND loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
                        THEN COALESCE(loan_disbursed_ugx, 0)
                        ELSE 0
                    END
                )
        END AS closed_prior_loans_principal_repayment_ratio

    FROM prior_loans_ranked
    GROUP BY target_disbursement_fid
),

-- ---------------------------------------------------------------------
-- Features that specifically require closed-loan ranking
-- ---------------------------------------------------------------------
closed_prior_loan_features AS (
    SELECT
        target_disbursement_fid,

        MAX(
            CASE
                WHEN closed_loan_recency <= 5
                THEN tenure_days
            END
        ) AS max_tenure_days_last_5_closed_loans

    FROM closed_prior_loans_ranked
    GROUP BY target_disbursement_fid
),

last_closed_loan_features AS (
    SELECT
        target_disbursement_fid,

        loan_uid AS last_closed_loan_uid,
        closure_date AS last_closed_loan_closure_date,
        tenure_days AS last_closed_loan_tenure_days,
        aging_bucket AS last_closed_loan_aging_bucket,

        loan_disbursed_ugx
            AS last_closed_loan_disbursed_ugx,

        loan_repaid_ugx
            AS last_closed_loan_repaid_ugx,

        loan_gross_repaid_ugx
            AS last_closed_loan_gross_repaid_ugx,

        CASE
            WHEN loan_disbursed_ugx > 0
            THEN loan_repaid_ugx / loan_disbursed_ugx
        END AS last_closed_loan_principal_repayment_ratio

    FROM closed_prior_loans_ranked
    WHERE closed_loan_recency = 1
),

-- ---------------------------------------------------------------------
-- Consecutive on-time closed-loan streak
--
-- tenure_days <= 1 is provisional and must be validated against the
-- source's timestamp/calendar-day semantics.
-- ---------------------------------------------------------------------
closed_loan_streak_rows AS (
    SELECT
        target_disbursement_fid,
        closed_loan_recency,
        tenure_days,

        SUM(
            CASE
                WHEN tenure_days > 1
                  OR tenure_days IS NULL
                THEN 1
                ELSE 0
            END
        ) OVER (
            PARTITION BY target_disbursement_fid
            ORDER BY closed_loan_recency
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS late_loans_seen

    FROM closed_prior_loans_ranked
),

on_time_streak AS (
    SELECT
        target_disbursement_fid,

        SUM(
            CASE
                WHEN tenure_days <= 1
                 AND late_loans_seen = 0
                THEN 1
                ELSE 0
            END
        ) AS consecutive_on_time_loans

    FROM closed_loan_streak_rows
    GROUP BY target_disbursement_fid
),

-- ---------------------------------------------------------------------
-- Prior disbursement frequency and amount
--
-- The intraday timestamp condition correctly includes loan 1 as prior
-- history for loan 2 on the same day.
-- ---------------------------------------------------------------------
prior_disbursement_features AS (
    SELECT
        current_loan.disbursement_fid
            AS target_disbursement_fid,

        COUNT(prior_loan.disbursement_fid)
            AS prior_disbursement_count,

        COUNT_IF(
            prior_loan.loan_date
                >= DATE_ADD('day', -30, current_loan.loan_date)
        ) AS prior_disbursement_count_30d,

        COUNT_IF(
            prior_loan.loan_date
                >= DATE_ADD('day', -90, current_loan.loan_date)
        ) AS prior_disbursement_count_90d,

        COUNT_IF(
            prior_loan.loan_date
                >= DATE_ADD('day', -180, current_loan.loan_date)
        ) AS prior_disbursement_count_180d,

        -- Thin-file routing signals: sufficient recent loan volume AND
        -- temporal breadth over the trailing 180 days, both point-in-time
        -- per target loan. Using disbursement_ts (not loan_date) for the
        -- window bound, matching the join condition below, correctly
        -- credits an earlier same-day disbursement as prior history for a
        -- later one on that same calendar date.
        COUNT_IF(
            prior_loan.disbursement_ts
                >= DATE_ADD('day', -180, current_loan.disbursement_ts)
        ) AS prior_loan_count_180d,

        COUNT(
            DISTINCT CASE
                WHEN prior_loan.disbursement_ts
                        >= DATE_ADD('day', -180, current_loan.disbursement_ts)
                THEN DATE_TRUNC('month', prior_loan.disbursement_ts)
            END
        ) AS prior_active_loan_months_180d,

        AVG(
            CASE
                WHEN prior_loan.loan_date
                    >= DATE_ADD('day', -30, current_loan.loan_date)
                THEN prior_loan.disbursement_amount_ugx
            END
        ) AS avg_prior_loan_amount_30d,

        AVG(
            CASE
                WHEN prior_loan.loan_date
                    >= DATE_ADD('day', -90, current_loan.loan_date)
                THEN prior_loan.disbursement_amount_ugx
            END
        ) AS avg_prior_loan_amount_90d,

        AVG(
            CASE
                WHEN prior_loan.loan_date
                    >= DATE_ADD('day', -180, current_loan.loan_date)
                THEN prior_loan.disbursement_amount_ugx
            END
        ) AS avg_prior_loan_amount_180d,

        COALESCE(
            SUM(
                CASE
                    WHEN prior_loan.loan_date
                        >= DATE_ADD('day', -30, current_loan.loan_date)
                    THEN prior_loan.disbursement_amount_ugx
                    ELSE 0
                END
            ),
            0
        ) AS prior_disbursed_ugx_30d,

        COALESCE(
            SUM(
                CASE
                    WHEN prior_loan.loan_date
                        >= DATE_ADD('day', -90, current_loan.loan_date)
                    THEN prior_loan.disbursement_amount_ugx
                    ELSE 0
                END
            ),
            0
        ) AS prior_disbursed_ugx_90d,

        COALESCE(
            SUM(
                CASE
                    WHEN prior_loan.loan_date
                        >= DATE_ADD('day', -180, current_loan.loan_date)
                    THEN prior_loan.disbursement_amount_ugx
                    ELSE 0
                END
            ),
            0
        ) AS prior_disbursed_ugx_180d,

        CASE
            WHEN COUNT(prior_loan.disbursement_fid) > 0
            THEN
                DATE_DIFF(
                    'month',
                    MIN(prior_loan.loan_date),
                    current_loan.loan_date
                ) + 1
        END AS months_since_first_observed_loan

    FROM hive.analytics.tmp_target_loans current_loan

    LEFT JOIN hive.analytics.tmp_disbursements prior_loan
      ON prior_loan.msisdn = current_loan.msisdn
     AND prior_loan.disbursement_ts
            < current_loan.disbursement_ts

    GROUP BY
        current_loan.disbursement_fid,
        current_loan.loan_date
),

-- ---------------------------------------------------------------------
-- Current-loan amount and timing features
-- ---------------------------------------------------------------------
current_loan_features AS (
    SELECT
        d.disbursement_fid,

        EXTRACT(HOUR FROM d.disbursement_ts)
            AS disbursement_hour,

        EXTRACT(DAY_OF_WEEK FROM d.disbursement_ts)
            AS disbursement_day_of_week,

        CASE
            WHEN EXTRACT(HOUR FROM d.disbursement_ts) >= 16
            THEN 1
            ELSE 0
        END AS late_afternoon_disbursement,

        CASE
            WHEN p.avg_prior_loan_amount_30d > 0
            THEN
                d.disbursement_amount_ugx
                / p.avg_prior_loan_amount_30d
        END AS current_to_avg_prior_loan_amount_30d,

        CASE
            WHEN p.avg_prior_loan_amount_90d > 0
            THEN
                d.disbursement_amount_ugx
                / p.avg_prior_loan_amount_90d
        END AS current_to_avg_prior_loan_amount_90d,

        CASE
            WHEN p.avg_prior_loan_amount_180d > 0
            THEN
                d.disbursement_amount_ugx
                / p.avg_prior_loan_amount_180d
        END AS current_to_avg_prior_loan_amount_180d,

        CASE
            WHEN p.months_since_first_observed_loan > 0
            THEN
                TRY_CAST(p.prior_disbursement_count AS DOUBLE)
                / p.months_since_first_observed_loan
        END AS prior_loans_per_observed_month

    FROM hive.analytics.tmp_target_loans d

    LEFT JOIN prior_disbursement_features p
      ON p.target_disbursement_fid = d.disbursement_fid
)
