-- ============================================================
-- ExtraFloat — Loan-Level Training Data Query  (v2)
-- Sources:
--   analytics.momo_loan_book_tracker_disbursements_daily
--   analytics.momo_loan_book_tracker_loan_state_daily
--
-- Grain: one row per (disbursement_fid, split).
--   Each loan belongs to exactly one split; the split date ranges are
--   disjoint, so the same disbursement_fid cannot appear in both train
--   and validation.
--
-- loan_status values in momo_loan_book_tracker_loan_state_daily:
--   OPEN_PRE_WINDOW  — loan was active before the Jan 2026 data window
--   OPEN             — normal active loan within 24h window
--   SETTLED          — fully repaid
--   CLOSED           — closed by system
--   OVERPAID         — agent paid more than outstanding
--   ANOMALY_OPEN     — agent took a NEW loan before settling the current one;
--                      the outstanding balance was transferred to the new loan.
--                      is_anomaly_open = 1.  Signals rollover / debt accumulation.
--   NEVER_BORROWED   — agent has no loan history (excluded from history ranking)
--
-- Label:
--   bad_state = 1 if in the 7 days AFTER disbursement (days 1–7, exclusive of
--   day 0 which is the disbursement day):
--     (a) MAX(days_aging) >= 1        — loan overdue after the 24h window, OR
--     (b) loan_status = 'ANOMALY_OPEN' — agent re-borrowed before settling;
--                                       balance was transferred, not repaid.
--   bad_state = 0 if neither condition was observed in the window.
--
--   NOTE: bad_state = 0 means "no overdue or rollover observed," not a confirmed
--   24h repayment.  That equivalence holds only if loan_state_daily has no gaps
--   for open loans and SETTLED/CLOSED means fully repaid.  Validate before use.
--
-- Features:
--   Loan behaviour strictly before each disbursement date (point-in-time).
--   Uses MAX(date_key) + join (not global ROW_NUMBER) so that every disbursement
--   — not just the agent's most recent — receives the correct historical snapshot.
--
-- Date arithmetic:
--   All window comparisons use horizon_date_key, derived from the DATE column
--   via YEAR() * 10000 + MONTH() * 100 + DAY(), to avoid month-end arithmetic
--   errors (e.g. 20260528 + 7 = 20260535, which is not a valid date).
--
-- Trino / Athena compatible.
-- ============================================================

WITH

-- ── Split definitions: disjoint date ranges, no overlap ──────────────────────
-- Adding more rows here adds more splits; never let date ranges overlap.
splits AS (
    SELECT DATE '2026-01-01' AS split_start, DATE '2026-04-30' AS split_end, 'train'      AS split
    UNION ALL
    SELECT DATE '2026-05-01' AS split_start, DATE '2026-07-31' AS split_end, 'validation' AS split
),

-- ── Step 1: One row per disbursement_fid ─────────────────────────────────────
-- momo_loan_book_tracker_disbursements_daily may contain one row per loan per
-- day (a daily snapshot), not a single immutable row per loan.  Deduplicate to
-- the earliest date_key to recover the original disbursement event.
disbursements_dedup AS (
    SELECT
        customer_msisdn                                             AS msisdn,
        disbursement_fid,
        CAST(disbursement_ts AS DATE)                              AS loan_date,
        disbursement_amount_ugx,
        sales_region,
        sales_territory,
        district,
        ROW_NUMBER() OVER (
            PARTITION BY disbursement_fid
            ORDER BY date_key ASC                    -- earliest row = original event
        )                                                          AS rn
    FROM analytics.momo_loan_book_tracker_disbursements_daily
    WHERE ova      = 'XTRAFLOAT-AGENT'
      AND date_key >= 20260101
),

-- Assign split membership and pre-compute YYYYMMDD integer keys for all
-- date-range comparisons against date_key in loan_state_daily.
-- Using YEAR()*10000 + MONTH()*100 + DAY() avoids the month-end integer
-- arithmetic error that occurs when adding a plain integer to 20260528, etc.
disbursements AS (
    SELECT
        d.msisdn,
        d.disbursement_fid,
        d.loan_date,
        s.split,
        d.disbursement_amount_ugx,
        d.sales_region,
        d.sales_territory,
        d.district,
        -- YYYYMMDD key for the disbursement day (day 0 — not yet due)
        YEAR(d.loan_date)  * 10000
            + MONTH(d.loan_date)  * 100
            + DAY(d.loan_date)                                     AS loan_date_key,
        -- YYYYMMDD key for day 7 after disbursement (end of outcome window)
        YEAR(DATE_ADD('day', 7, d.loan_date))  * 10000
            + MONTH(DATE_ADD('day', 7, d.loan_date)) * 100
            + DAY(DATE_ADD('day', 7, d.loan_date))                 AS horizon_date_key
    FROM disbursements_dedup d
    JOIN splits s ON d.loan_date BETWEEN s.split_start AND s.split_end
    WHERE d.rn = 1                                                 -- one row per loan
),

-- ── Step 2: Label — outcome in days 1–7 after disbursement ───────────────────
-- Outcome window: date_key strictly AFTER disbursement day (> loan_date_key)
-- through day 7 inclusive (<= horizon_date_key).
-- Day 0 is excluded: the 24h due date occurs at 24h after disbursement_ts,
-- which falls within day 1 of the state table.
--
-- bad_state = 1 if either:
--   (a) any state row shows days_aging >= 1 (loan overdue past the 24h window)
--   (b) any state row shows ANOMALY_OPEN (agent re-borrowed, balance transferred)
--
-- Uniqueness assumption: (customer_msisdn, current_loan_start_date) identifies
-- a unique loan because XtraFloat permits at most one active loan at a time.
-- Edge case: two separate loans disbursed to the same agent on the same calendar
-- date would both match here (over-count).  Detectable post-hoc by checking
-- disbursement_event_count > 1 in loan_state_daily for the affected date.
loan_labels AS (
    SELECT
        d.msisdn,
        d.disbursement_fid,
        d.split,
        MAX(ls.days_aging)                                                        AS max_days_aging_7d,
        CASE
            WHEN MAX(ls.days_aging) >= 1
              OR MAX(CASE WHEN ls.loan_status = 'ANOMALY_OPEN' THEN 1 ELSE 0 END) = 1
            THEN 1 ELSE 0
        END                                                                       AS bad_state,
        COUNT(DISTINCT ls.date_key)                                               AS observed_days,
        MIN(ls.date_key)                                                          AS first_obs_key,
        MAX(ls.date_key)                                                          AS last_obs_key
    FROM disbursements d
    JOIN analytics.momo_loan_book_tracker_loan_state_daily ls
        ON  ls.customer_msisdn          = d.msisdn
        AND ls.current_loan_start_date  = d.loan_date         -- same loan
        AND ls.date_key                 > d.loan_date_key     -- day 1 onward (past 24h window)
        AND ls.date_key                <= d.horizon_date_key  -- through day 7
    GROUP BY d.msisdn, d.disbursement_fid, d.split
),

-- ── Step 3a: Point-in-time date lookup ───────────────────────────────────────
-- For each disbursement, find the most-recent date_key in loan_state_daily
-- that is strictly before the disbursement date.
--
-- Why MAX(date_key) + join instead of ROW_NUMBER:
--   ROW_NUMBER() OVER (PARTITION BY msisdn ORDER BY date_key DESC) assigns rn=1
--   to the agent's globally most-recent row across all history.  Joining on
--   rn = 1 AND date_key < loan_date_key returns NULL for every disbursement that
--   is not the agent's last, because the rn=1 row (later in time) fails the
--   date_key < loan_date_key filter.  This leaves all but the final disbursement
--   without historical features — silently and incorrectly.
--   The MAX + join pattern resolves this correctly per disbursement.
pit_lookup AS (
    SELECT
        d.msisdn,
        d.disbursement_fid,
        d.split,
        d.loan_date,                                           -- carried for days_since_last_repayment
        MAX(ls.date_key)                                       AS pit_date_key
    FROM disbursements d
    JOIN analytics.momo_loan_book_tracker_loan_state_daily ls
        ON  ls.customer_msisdn = d.msisdn
        AND ls.date_key        < d.loan_date_key               -- strictly before this disbursement
        AND ls.loan_status    != 'NEVER_BORROWED'              -- no loan history — skip
    GROUP BY d.msisdn, d.disbursement_fid, d.split, d.loan_date
),

-- ── Step 3b: Fetch the feature row at the point-in-time date ─────────────────
-- Join back to loan_state_daily on the exact pit_date_key found above.
-- All lifetime_* columns are cumulative running totals, so this single row
-- captures all historical behaviour up to (but not including) this loan.
point_in_time AS (
    SELECT
        p.msisdn,
        p.disbursement_fid,
        p.split,
        ls.loan_seq,
        -- has_pre_window_history is boolean in source; CASE WHEN avoids cast ambiguity
        CASE WHEN ls.has_pre_window_history THEN 1 ELSE 0 END           AS has_pre_window_history,
        -- Non-zero means the prior loan was not fully settled when this one was taken
        ls.outstanding_ugx,
        -- Repayment duration of the most-recent closed loan
        ls.tenure_days                                                   AS last_tenure_days,
        -- Cumulative repayment ratio across all prior loans
        CASE
            WHEN ls.lifetime_disbursed_ugx > 0
            THEN TRY_CAST(ls.lifetime_repaid_ugx AS DOUBLE)
                 / TRY_CAST(ls.lifetime_disbursed_ugx AS DOUBLE)
            ELSE NULL
        END                                                              AS lifetime_repayment_ratio,
        -- 1 = the prior loan rolled over into a new one (ANOMALY_OPEN) without settlement
        CASE WHEN ls.loan_status = 'ANOMALY_OPEN' THEN 1 ELSE 0 END    AS had_prior_rollover,
        -- Days between the last repayment and this disbursement (recency of good behaviour)
        DATE_DIFF('day', ls.last_repayment_date, p.loan_date)           AS days_since_last_repayment
    FROM pit_lookup p
    JOIN analytics.momo_loan_book_tracker_loan_state_daily ls
        ON  ls.customer_msisdn = p.msisdn
        AND ls.date_key        = p.pit_date_key
)

-- ── Final output ─────────────────────────────────────────────────────────────
-- Observability filter: require at least 5 of 7 outcome days to be present.
-- This is a pragmatic threshold for loans near the end of available data.
-- A stronger alternative (once max_data_date is known):
--   AND d.loan_date <= [max_data_date in loan_state_daily] - INTERVAL '7' DAY
-- which guarantees the full 7-day window exists in the warehouse.
SELECT
    d.msisdn,
    d.disbursement_fid,
    d.split,
    d.loan_date,
    d.disbursement_amount_ugx,
    d.sales_region,
    d.sales_territory,
    d.district,

    -- ── Historical features (point-in-time, no leakage) ──────────────────
    -- NULL means the agent had no prior loan history in the data window.
    -- First-loan agents (loan_seq = 0) will have all feature columns NULL or 0.
    COALESCE(p.loan_seq,                 0)     AS loan_seq,
    COALESCE(p.has_pre_window_history,   0)     AS has_pre_window_history,
    COALESCE(p.outstanding_ugx,          0)     AS outstanding_at_disburse,
    p.last_tenure_days,                          -- NULL for first-loan agents
    p.lifetime_repayment_ratio,                  -- NULL for first-loan agents
    COALESCE(p.had_prior_rollover,       0)     AS had_prior_rollover,
    p.days_since_last_repayment,                 -- NULL for first-loan agents

    -- ── Label ─────────────────────────────────────────────────────────────
    l.max_days_aging_7d,
    l.bad_state,
    l.observed_days,
    l.first_obs_key,
    l.last_obs_key

FROM disbursements d
LEFT JOIN point_in_time p
    ON  p.msisdn           = d.msisdn
    AND p.disbursement_fid = d.disbursement_fid
    AND p.split            = d.split
LEFT JOIN loan_labels l
    ON  l.msisdn           = d.msisdn
    AND l.disbursement_fid = d.disbursement_fid
    AND l.split            = d.split

WHERE l.bad_state    IS NOT NULL           -- loan must appear in state table (label exists)
  AND l.observed_days >= 5                 -- at least 5 of 7 outcome days observed
