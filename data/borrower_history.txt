-- ============================================================================
-- borrower_history.txt
-- ============================================================================
-- Produces the BORROWER-LEVEL dataset expected by
-- prepare_borrower_limit_features() / load_borrower_limit_features()
-- (one row per phonenumber).
--
-- Sourced from the MoMo loan-book tracker tables (replaces the old
-- devdata.xtrafloat_daily_trans transaction log):
--   analytics.momo_loan_book_tracker_disbursements_daily  -- one row per
--     disbursement, keyed by disbursement_fid.
--   analytics.momo_loan_book_tracker_repayments_daily     -- one row per
--     repayment (atomic transactions). Carries no disbursement_fid, so each
--     repayment is attributed to whichever loan was open for that customer
--     at the time of payment -- see repay_attributed below. This is a proxy
--     for the exact requestid match the old transaction log provided.
--   analytics.momo_loan_book_tracker_loan_state_daily     -- daily snapshot
--     of loan state, keyed by disbursement_fid. Used to (a) detect loans
--     that were EVER ANOMALY_OPEN at any point in their history (see
--     ever_anomaly_open in checkpoint 1) and correct their total_recovered/
--     total_cure_cashflow to the authoritative lifetime_repaid_ugx -- NOT to
--     exclude them from the population (a loan flagged ANOMALY_OPEN is not a
--     system error: its own lifetime_repaid_ugx keeps climbing normally and
--     it typically resolves to CLOSED within days -- confirmed via
--     data/anomaly_open_balance_transfer_test.sql; the population is instead
--     gated on days_aging so only a loan stuck ANOMALY_OPEN well past
--     normal resolution time is excluded -- see loan_state_anomalies below),
--     and (b) enrich the output with authoritative lifecycle/charge fields
--     for the borrower's latest loan.
--
-- EXECUTION: this file is NOT a single runnable statement -- it is split by
-- the ##BORROWER_HISTORY_CHECKPOINT_0##/_1##/_2## markers below into four
-- SQL statements (three CREATE TABLE ... AS SELECT checkpoints
-- materializing the per-loan-grain pipeline in three stages, then a
-- CREATE OR REPLACE VIEW building the borrower-level output from the last
-- checkpoint), because the unsplit query -- and even each of the first two
-- split attempts -- exceeded the warehouse's query-plan stage-count
-- ceiling. Use scripts/build_vw_bh_output.py <schema> to generate all four
-- statements with the checkpoint table names and schema substituted in --
-- see the comments at each checkpoint marker for why. Do not hand-paste this
-- file as one query.
--
-- Self-contained: the per-loan-grain pipeline below (through the checkpoint
-- marker) replaces the old transaction-log sourcing. loan_core through
-- exposure_features is otherwise unchanged from
-- the previous version of this file -- none of the intermediate fields
-- (total_penalty, total_reimbursed, loan_status, repayment_bucket,
-- outstanding_balance) reach the final output; only requestid/phonenumber/
-- event_ts/amount/txn_type feed them, and those five fields are reproduced
-- exactly by the base/dedup/classified section. The one deliberate behavior
-- change: on_time_24h_flag/on_time_26h_flag/default_24h_flag/
-- default_26h_flag/severe_default_48h_flag in loan_final are now
-- maturity-gated (NULL until the loan has actually had 24h/26h/48h to
-- cure, per snapshot_ts below) rather than treating an unseasoned loan as a
-- default -- inherited unchanged from the original query until this fix.
-- default_rate_last_50_loans in `streaks` was adjusted accordingly (see
-- mature_loans_last_50 in recent_loan_windows). The streak-group/
-- streak-current logic elsewhere in `streaks`/`streaks_final` needed no
-- change -- verified that NULL flags already propagate through those SUM/
-- CASE expressions as a correct no-op (an unseasoned loan neither breaks
-- nor extends a streak).
-- A second deliberate behavior change: loan_state_anomalies (checkpoint 0)
-- no longer excludes a loan purely because its LATEST snapshot shows
-- ANOMALY_OPEN -- that was based on a "system error/balance transfer"
-- theory since confirmed false, and was unstable (the same loan flipped
-- included/excluded across re-runs depending on incidental query-run
-- timing relative to its own resolution). It is now gated on days_aging, so
-- a freshly-flagged loan -- overwhelmingly likely to resolve within days --
-- stays in the population, and only a loan stuck ANOMALY_OPEN well past
-- normal resolution time is excluded. Separately, loan_final now computes
-- ever_anomaly_open (from the full loan_state_daily history, not just the
-- latest snapshot) and overrides total_recovered/total_cure_cashflow with
-- the authoritative lifetime_repaid_ugx for any loan that was ever
-- ANOMALY_OPEN, correcting a confirmed attribution-window misattribution
-- (repayments landing after a second loan's disbursement were being
-- credited to the wrong loan by the phonenumber-window heuristic below).
-- The cure-timing flags (on_time_24h_flag etc.) are NOT corrected by this
-- change -- see the KNOWN LIMITATION comment near their definition.
-- ============================================================================
WITH snapshots AS (
-- as_of_load_ts bounds inserted_ts across all three source tables so that a
-- rerun of this exact snapshot is reproducible: dedup previously took only
-- "latest inserted_ts" with no upper bound, so a late backfill/correction
-- landing after the original run could silently change a re-run of "the
-- same" historical snapshot. Set as_of_load_ts to the inserted_ts ceiling
-- of the run you want to reproduce (e.g. the max inserted_ts observed when
-- the original extract was produced); use CURRENT_TIMESTAMP only when you
-- deliberately want the freshest reprocessed data instead of a frozen replay.
--
-- snapshot_ts is the business observation time used to decide whether a
-- loan has had enough time to actually earn an on-time/default label (see
-- on_time_24h_flag etc. in loan_final below) -- a loan disbursed shortly
-- before snapshot_ts hasn't had its 24h/26h/48h grace period elapse yet, so
-- it isn't a default, it's simply not observable yet. Distinct from
-- as_of_load_ts (which bounds what data has loaded, not what time "now" is
-- for maturity purposes), though the two commonly coincide in practice. Set
-- it on or after the end of snapshot_dt's calendar day; for the same
-- reproducibility reason as as_of_load_ts, use a frozen literal, not
-- CURRENT_TIMESTAMP, when replaying a historical snapshot.
SELECT
cast(20260731 AS bigint) AS snapshot_dt,
cast('2026-08-20 00:00:00.000' AS timestamp) AS as_of_load_ts,
cast('2026-08-20 00:00:00.000' AS timestamp) AS snapshot_ts
),
-- ── Begin: loan-level query sourced from the momo_loan_book_tracker tables ──
-- DEDUP-VS-SCOPE ORDERING (applies here and identically to repay_raw and
-- loan_state_loads_dedup below): the business-date filter (date(...) <=
-- snapshot_dt) is applied BEFORE deduplication picks the latest inserted_ts
-- version. That means the implemented semantics are "latest IN-SCOPE
-- record as of load cutoff" -- not "latest record as of load cutoff, then
-- apply business-date scope." If a correction changes an event's own
-- timestamp (e.g. disbursement_ts) to move it outside snapshot_dt, this
-- filter drops that latest version entirely and dedup falls back to an
-- older, possibly-superseded in-scope version, rather than recognizing the
-- event moved out of scope. This is safe only if disbursement_ts/
-- repayment_ts are immutable per ID once written -- confirm that guarantee
-- with the source system owner; if event timestamps can be corrected, this
-- ordering needs to be revisited deliberately, not assumed.
disb_raw AS (
-- ova = 'XTRAFLOAT-AGENT' -- all three momo_loan_book_tracker_* tables
-- carry activity from OTHER MoMo services too, not just XtraFloat agent
-- lending, confirmed directly against the warehouse. Without this filter,
-- disbursements/repayments/loan-state rows from unrelated services leak in
-- for any msisdn that also uses those other services -- this was traced
-- back as the likely explanation for several apparent anomalies found via
-- data/spot_check_loan_trace.sql (a repayment_fid appearing to cover
-- multiple unrelated amounts, a loan_state_daily lifetime_disbursed_ugx
-- that didn't match disbursements_daily's own disbursement_amount_ugx for
-- the same disbursement_fid). Applied to all three tables' raw CTEs below.
-- date_key <= snapshot_dt is a PARTITION filter (this table is partitioned
-- by date_key, confirmed directly against the warehouse) -- without it,
-- every query here scans every partition regardless of how selective the
-- other predicates are. Kept alongside the existing date(disbursement_ts)
-- business-date filter, not instead of it: this assumes date_key tracks
-- disbursement_ts's own calendar date closely enough to use as a safe
-- upper bound, mirroring how loan_state_loads_dedup below has always used
-- date_key. If date_key ever represents something else (e.g. load/batch
-- date rather than event date), this bound could wrongly exclude in-scope
-- rows -- spot-check with data/spot_check_loan_trace.sql before trusting
-- this filter in production if that assumption hasn't been confirmed.
SELECT
disbursement_fid,
regexp_replace(trim(cast(customer_msisdn AS varchar)), '[^0-9]', '') AS phonenumber,
try_cast(disbursement_ts AS timestamp) AS event_ts,
cast(disbursement_amount_ugx AS double) AS raw_amount,
inserted_ts
FROM analytics.momo_loan_book_tracker_disbursements_daily
WHERE ova = 'XTRAFLOAT-AGENT'
AND try_cast(disbursement_ts AS timestamp) IS NOT NULL
AND disbursement_fid IS NOT NULL
AND customer_msisdn IS NOT NULL
AND date_key <= (SELECT snapshot_dt FROM snapshots)
AND date(try_cast(disbursement_ts AS timestamp)) <= date_parse(cast((SELECT snapshot_dt FROM snapshots) AS varchar), '%Y%m%d')
AND inserted_ts <= (SELECT as_of_load_ts FROM snapshots)
),
disb_dedup AS (
SELECT *
FROM (
SELECT d.*,
ROW_NUMBER() OVER (
PARTITION BY disbursement_fid
ORDER BY inserted_ts DESC
) rn
FROM disb_raw d
)
WHERE rn = 1
),
disb_windows AS (
-- Per-customer disbursement listing, renamed for downstream clarity.
-- next_disbursement_ts (a LEAD() window function requiring its own
-- phonenumber-keyed shuffle) used to live here for repay_attributed's old
-- range-join attribution -- removed since repay_attributed below now
-- derives the same "bounded by the next disbursement" semantics via a
-- forward-fill window function instead, which needs only each
-- disbursement's own timestamp, not a precomputed next_disbursement_ts.
-- Confirmed unused elsewhere in this file before removing.
SELECT
disbursement_fid,
phonenumber,
event_ts AS disbursement_ts,
raw_amount AS disbursed_amount
FROM disb_dedup
),
repay_raw AS (
-- date_key <= snapshot_dt: same partition-pruning rationale/assumption as
-- disb_raw above.
SELECT
repayment_fid,
regexp_replace(trim(cast(customer_msisdn AS varchar)), '[^0-9]', '') AS phonenumber,
try_cast(repayment_ts AS timestamp) AS event_ts,
cast(repayment_amount_ugx AS double) AS raw_amount,
inserted_ts
FROM analytics.momo_loan_book_tracker_repayments_daily
WHERE ova = 'XTRAFLOAT-AGENT'
AND try_cast(repayment_ts AS timestamp) IS NOT NULL
AND repayment_fid IS NOT NULL
AND customer_msisdn IS NOT NULL
AND date_key <= (SELECT snapshot_dt FROM snapshots)
AND date(try_cast(repayment_ts AS timestamp)) <= date_parse(cast((SELECT snapshot_dt FROM snapshots) AS varchar), '%Y%m%d')
AND inserted_ts <= (SELECT as_of_load_ts FROM snapshots)
),
repay_dedup AS (
SELECT *
FROM (
SELECT r.*,
ROW_NUMBER() OVER (
PARTITION BY repayment_fid
ORDER BY inserted_ts DESC
) rn
FROM repay_raw r
)
WHERE rn = 1
),
attribution_timeline AS (
-- Forward-fill attribution instead of a phonenumber + timestamp-range
-- JOIN: interleave each phonenumber's disbursements and repayments into
-- one sorted timeline, then carry the most recent disbursement_fid
-- forward onto each repayment (attribution_filled below). Logically
-- identical to the old range join (r.event_ts >= disbursement_ts AND
-- r.event_ts < next_disbursement_ts) but costs one sort per phonenumber
-- instead of an O(disbursements x repayments) pairwise comparison within
-- each phonenumber's bucket -- the same join shape that caused the
-- validation suite's B2b/B3 600GB memory failures (see
-- borrower_history_validation_queries.sql). is_repayment orders
-- same-instant disbursement/repayment ties so the repayment picks up the
-- JUST-disbursed loan, matching the old join's inclusive >= on the
-- disbursement side and exclusive < on the next-disbursement side;
-- disbursement_fid is a second tiebreak for the rare case of two
-- disbursements at the identical instant (Section B4 in
-- borrower_history_validation_queries.sql), matching the removed
-- LEAD(...) ORDER BY event_ts, disbursement_fid exactly.
SELECT
phonenumber,
disbursement_ts AS event_ts,
disbursed_amount AS raw_amount,
0 AS is_repayment,
disbursement_fid
FROM disb_windows
UNION ALL
SELECT
phonenumber,
event_ts,
raw_amount,
1 AS is_repayment,
NULL AS disbursement_fid
FROM repay_dedup
),
attribution_filled AS (
SELECT
phonenumber,
event_ts,
raw_amount,
is_repayment,
LAST_VALUE(disbursement_fid) IGNORE NULLS OVER (
PARTITION BY phonenumber
ORDER BY event_ts, is_repayment, disbursement_fid
ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
) AS requestid
FROM attribution_timeline
),
repay_attributed AS (
-- Attribute each repayment to the most recent prior disbursement for that
-- customer, bounded by the next disbursement (if any) -- see
-- attribution_timeline/attribution_filled above for how. Repayments with
-- no matching prior disbursement in the query window are dropped
-- (requestid IS NULL), same semantics as the old INNER JOIN.
SELECT
requestid,
phonenumber,
event_ts,
raw_amount
FROM attribution_filled
WHERE is_repayment = 1
AND requestid IS NOT NULL
),
loan_state_loads_dedup AS (
-- loan_state_daily is a daily snapshot; dedup re-ingested rows for the
-- same (disbursement_fid, date_key) before picking the as-of snapshot.
SELECT *
FROM (
SELECT lsd.*,
ROW_NUMBER() OVER (
PARTITION BY disbursement_fid, date_key
ORDER BY inserted_ts DESC
) rn
FROM analytics.momo_loan_book_tracker_loan_state_daily lsd
WHERE ova = 'XTRAFLOAT-AGENT'
AND disbursement_fid IS NOT NULL
AND date_key <= (SELECT snapshot_dt FROM snapshots)
AND inserted_ts <= (SELECT as_of_load_ts FROM snapshots)
)
WHERE rn = 1
),
loan_state_snapshot AS (
-- One row per disbursement_fid: its latest known state at/before the
-- snapshot cutoff. A loan still open at the cutoff yields its latest
-- state; a loan already closed yields its final closing-day row (rows
-- stop appearing in loan_state_daily once a loan closes), which is
-- exactly "state as of the snapshot" either way.
SELECT *
FROM (
SELECT lsld.*,
ROW_NUMBER() OVER (
PARTITION BY disbursement_fid
ORDER BY date_key DESC
) rn2
FROM loan_state_loads_dedup lsld
)
WHERE rn2 = 1
),
loan_state_anomalies AS (
-- POPULATION-STABILITY FIX: previously excluded ANY current ANOMALY_OPEN
-- loan, based on a "system error" theory since confirmed false (an
-- ANOMALY_OPEN loan's own lifetime_repaid_ugx keeps climbing normally and
-- it typically resolves to CLOSED within 1-6 days -- see
-- data/anomaly_open_raw_timeline_examples.sql,
-- data/anomaly_open_balance_transfer_test.sql). That current-status-only
-- exclusion was unstable: the same loan flipped between included/excluded
-- across re-runs purely from incidental query-run timing relative to its
-- own resolution. Gated on days_aging so normal, fast-resolving cases stay
-- IN the population; only a loan still stuck in ANOMALY_OPEN well past the
-- time a normal case takes to resolve is treated as a genuine long-running
-- anomaly and excluded. Threshold of 7 days matches this project's
-- existing aging-bucket convention (see
-- data/anomaly_open_aging_only_test.sql,
-- data/silently_stuck_loans_characterization.sql); tune if warehouse data
-- shows normal resolution routinely exceeding this.
SELECT disbursement_fid
FROM loan_state_snapshot
WHERE (loan_status = 'ANOMALY_OPEN' OR is_anomaly_open = true)
AND days_aging > 7
),
classified AS (
SELECT
w.disbursement_fid AS requestid,
w.phonenumber,
w.disbursement_ts AS event_ts,
abs(w.disbursed_amount) AS amount,
'disbursement' AS txn_type
FROM disb_windows w
WHERE w.disbursement_fid NOT IN (SELECT disbursement_fid FROM loan_state_anomalies)
UNION ALL
SELECT
ra.requestid,
ra.phonenumber,
ra.event_ts,
abs(ra.raw_amount) AS amount,
'recovery' AS txn_type
FROM repay_attributed ra
WHERE ra.requestid NOT IN (SELECT disbursement_fid FROM loan_state_anomalies)
)
SELECT * FROM classified
-- ##BORROWER_HISTORY_CHECKPOINT_0##
-- ============================================================================
-- MATERIALIZATION CHECKPOINT 0 -- checkpoint 1 (further down) was STILL
-- hitting the warehouse's stage-count ceiling on its own (140 stages,
-- observed; cap is 100) even after checkpoints 2 and 1 (see their comments)
-- already split the original single-statement query twice. classified's
-- own UNION ALL + two NOT IN(loan_state_anomalies) exclusions, combined
-- with the cure-timing cascade and loan_final's 4-way trailing join, add up
-- to more than expected from window-function count alone -- Trino's stage
-- count scales with joins/unions/CTE boundaries too, not just OVER()
-- clauses. classified (transaction grain: one row per disbursement or
-- attributed repayment, ANOMALY_OPEN already excluded) is a clean, natural
-- checkpoint: materialize it here, then everything below re-derives
-- loan_state_snapshot fresh (cheap, 2 window functions -- not worth a
-- third physical table) but reads classified from the checkpoint table
-- instead of re-deriving the UNION/exclusion logic.
--
-- snapshots is redefined here too -- it's a plain literal-valued CTE (no
-- table scan), but it was only defined once, in checkpoint 0's own WITH
-- clause, and checkpoint 1 is now a separate SQL statement that can't see
-- it. Without this redefinition, every `(SELECT ... FROM snapshots)`
-- reference below (loan_state_loads_dedup's date_key/inserted_ts bounds,
-- and the on_time/default flags further down that compare against
-- snapshot_ts) resolves `snapshots` as an unqualified table lookup instead
-- of a CTE, which fails with "Schema must be specified when session schema
-- is not set" the moment no default session schema is configured.
WITH snapshots AS (
SELECT
cast(20260731 AS bigint) AS snapshot_dt,
cast('2026-08-20 00:00:00.000' AS timestamp) AS as_of_load_ts,
cast('2026-08-20 00:00:00.000' AS timestamp) AS snapshot_ts
),
loan_state_loads_dedup AS (
-- loan_state_daily is a daily snapshot; dedup re-ingested rows for the
-- same (disbursement_fid, date_key) before picking the as-of snapshot.
SELECT *
FROM (
SELECT lsd.*,
ROW_NUMBER() OVER (
PARTITION BY disbursement_fid, date_key
ORDER BY inserted_ts DESC
) rn
FROM analytics.momo_loan_book_tracker_loan_state_daily lsd
WHERE ova = 'XTRAFLOAT-AGENT'
AND disbursement_fid IS NOT NULL
AND date_key <= (SELECT snapshot_dt FROM snapshots)
AND inserted_ts <= (SELECT as_of_load_ts FROM snapshots)
)
WHERE rn = 1
),
loan_state_snapshot AS (
-- One row per disbursement_fid: its latest known state at/before the
-- snapshot cutoff. A loan still open at the cutoff yields its latest
-- state; a loan already closed yields its final closing-day row (rows
-- stop appearing in loan_state_daily once a loan closes), which is
-- exactly "state as of the snapshot" either way.
SELECT *
FROM (
SELECT lsld.*,
ROW_NUMBER() OVER (
PARTITION BY disbursement_fid
ORDER BY date_key DESC
) rn2
FROM loan_state_loads_dedup lsld
)
WHERE rn2 = 1
),
loan_ever_anomaly_open AS (
-- Per-loan flag: was this disbursement_fid EVER ANOMALY_OPEN at ANY
-- date_key in its own history -- not just its LATEST snapshot (all
-- loan_state_snapshot exposes). Computed from loan_state_loads_dedup's
-- one-row-per-(disbursement_fid, date_key) history so a loan that has since
-- resolved to CLOSED still carries the fact that it passed through
-- ANOMALY_OPEN. Used ONLY to correct total_recovered/total_cure_cashflow in
-- loan_final below -- population exclusion is handled separately by the
-- aging-gated loan_state_anomalies in checkpoint 0.
SELECT
disbursement_fid,
bool_or(loan_status = 'ANOMALY_OPEN' OR is_anomaly_open = true) AS ever_anomaly_open
FROM loan_state_loads_dedup
GROUP BY disbursement_fid
),
loan_core AS (
-- Combines what used to be two separate GROUP BY requestid, phonenumber
-- passes over {{CHECKPOINT_TABLE_0}} -- this CTE's own aggregates, plus a
-- former standalone recovery_times CTE -- into one. Same source table,
-- same grouping key, only the aggregate expressions differed, so scanning
-- and shuffling it twice was pure waste. first_recovery_ts/last_recovery_ts
-- (used by loan_final for days_to_last_recovery/hours_to_first_recovery/
-- hours_to_last_recovery) are now columns on loan_core directly, dropping
-- one of loan_final's four trailing LEFT JOINs.
SELECT
requestid,
phonenumber,
MIN(CASE WHEN txn_type='disbursement' THEN event_ts END) AS disbursement_ts,
SUM(CASE WHEN txn_type='disbursement' THEN amount ELSE 0 END) AS disbursed_amount,
SUM(CASE WHEN txn_type='recovery' THEN amount ELSE 0 END) AS total_recovered,
SUM(CASE WHEN txn_type='reimbursement' THEN amount ELSE 0 END) AS total_reimbursed,
SUM(CASE WHEN txn_type='penalty' THEN amount ELSE 0 END) AS total_penalty,
MIN(CASE WHEN txn_type IN ('recovery','reimbursement') THEN event_ts END) AS first_recovery_ts,
MAX(CASE WHEN txn_type IN ('recovery','reimbursement') THEN event_ts END) AS last_recovery_ts
FROM {{CHECKPOINT_TABLE_0}}
GROUP BY requestid, phonenumber
),
cure_events AS (
SELECT
c.requestid,
c.phonenumber,
c.event_ts,
CASE WHEN c.txn_type IN ('recovery','reimbursement') THEN c.amount ELSE 0 END AS cure_amount,
l.disbursement_ts,
l.disbursed_amount,
l.total_penalty
FROM {{CHECKPOINT_TABLE_0}} c
JOIN loan_core l
ON c.requestid = l.requestid
AND c.phonenumber = l.phonenumber
WHERE c.event_ts >= l.disbursement_ts
),
-- EXTENDED CURE PROGRESS
cure_progress_base AS (
SELECT
ce.*,
SUM(cure_amount) OVER (
PARTITION BY requestid, phonenumber
ORDER BY event_ts
) AS cumulative_cure,
-- CAST(... AS DOUBLE) is required, not cosmetic: date_diff(...) is BIGINT,
-- and Trino types the unsuffixed literal 3600.0 as DECIMAL(5,1) --
-- BIGINT/DECIMAL division follows SQL decimal-arithmetic scale-inference
-- rules and can silently truncate precision below what true floating-point
-- division carries. Confirmed live (data/24h_decimal_division_test.sql):
-- an elapsed time of 86,493 seconds (24.0258... true hours) divided this
-- way compared as <= 24 -- FALSE positive on "within 24h" -- while the
-- CAST(...AS DOUBLE) version correctly computed 24.0258333333 and compared
-- as > 24. Without this cast, recovery_6h/12h/24h/26h below can include a
-- cure event that happened AFTER the checkpoint they're named for, which
-- was confirmed to make default_24h_flag disagree with the independently
-- (and correctly) derived on_time_24h_flag for ~23% of borrowers
-- (GATE 1b's bad_24h_complement_identity in
-- borrower_history_validation_queries.sql).
CAST(date_diff('second', disbursement_ts, event_ts) AS DOUBLE) / 3600.0 AS hours_since_disbursement,
COUNT(*) OVER (
PARTITION BY requestid, phonenumber
) AS txn_count,
SUM(CASE WHEN cure_amount > 0 THEN 1 ELSE 0 END) OVER (
PARTITION BY requestid, phonenumber
) AS recovery_count
FROM cure_events ce
),
cure_progress AS (
SELECT
cpb.*,
MAX(cumulative_cure) OVER (
PARTITION BY requestid, phonenumber
) AS max_cumulative_recovery
FROM cure_progress_base cpb
),
-- EXTENDED MILESTONES
-- =========================
-- EXTENDED MILESTONES + CHECKPOINTS (merged)
-- =========================
milestones AS (
-- Combines what used to be two separate GROUP BY requestid, phonenumber
-- passes over cure_progress -- this CTE's own milestone timestamps, plus a
-- former standalone checkpoints CTE's recovery_*h amounts -- into one.
-- Same source, same grouping key, only the aggregate expressions differed.
-- Dropping the standalone checkpoints CTE also removes one of loan_final's
-- four trailing LEFT JOINs.
SELECT
requestid,
phonenumber,
MAX(cumulative_cure) AS total_cure_cashflow,
MIN(CASE WHEN cumulative_cure >= disbursed_amount * 0.5 THEN event_ts END) AS ts_50pct,
MIN(CASE WHEN cumulative_cure >= disbursed_amount * 0.75 THEN event_ts END) AS ts_75pct,
MIN(CASE WHEN cumulative_cure >= disbursed_amount THEN event_ts END) AS principal_cure_ts,
MIN(CASE WHEN cumulative_cure >= disbursed_amount + total_penalty THEN event_ts END) AS full_balance_cure_ts,
COALESCE(MAX(cumulative_cure) FILTER (WHERE hours_since_disbursement <= 6),0) AS recovery_6h,
COALESCE(MAX(cumulative_cure) FILTER (WHERE hours_since_disbursement <= 12),0) AS recovery_12h,
COALESCE(MAX(cumulative_cure) FILTER (WHERE hours_since_disbursement <= 24),0) AS recovery_24h,
COALESCE(MAX(cumulative_cure) FILTER (WHERE hours_since_disbursement <= 26),0) AS recovery_26h
FROM cure_progress
GROUP BY requestid, phonenumber
),
-- EXTENDED LOAN FINAL
loan_final AS (
SELECT
l.requestid,
l.phonenumber,
l.disbursement_ts,
l.disbursed_amount,
-- ANOMALY-ATTRIBUTION FIX: for a loan that was EVER ANOMALY_OPEN
-- (ever_anomaly_open below), the phonenumber-window repayment-attribution
-- heuristic (attribution_timeline/attribution_filled in checkpoint 0)
-- misattributes repayments landing after a SECOND loan's disbursement to
-- that second loan instead of this one, even though the source system
-- correctly keeps crediting THIS loan's own lifetime_repaid_ugx. Confirmed
-- empirically: ever-ANOMALY_OPEN loans show a 70.3% vs 61.6% exact-match
-- rate against lifetime_repaid_ugx ground truth (~8.7-point gap, ~17% of
-- the surviving population -- see data/anomaly_open_b2b_correlation_test.sql).
-- For these loans, trust loan_state_daily's own authoritative
-- lifetime_repaid_ugx (ls_lifetime_repaid_ugx below) over the
-- attribution-derived total from {{CHECKPOINT_TABLE_0}}; COALESCE falls
-- back to the attribution-derived figure if lifetime_repaid_ugx itself is
-- NULL (e.g. the loan has no loan_state_daily row at all).
CASE
WHEN COALESCE(eao.ever_anomaly_open, false)
THEN COALESCE(ls.lifetime_repaid_ugx, l.total_recovered, 0)
ELSE COALESCE(l.total_recovered, 0)
END AS total_recovered,
COALESCE(l.total_reimbursed,0) AS total_reimbursed,
COALESCE(l.total_penalty,0) AS total_penalty,
CASE
WHEN COALESCE(eao.ever_anomaly_open, false)
THEN COALESCE(ls.lifetime_repaid_ugx, l.total_recovered, 0)
ELSE COALESCE(l.total_recovered, 0)
END + COALESCE(l.total_reimbursed,0) AS total_cure_cashflow,
m.principal_cure_ts,
m.full_balance_cure_ts,
-- CAST(... AS DOUBLE) required for the same reason as cure_progress_base's
-- hours_since_disbursement above -- see that comment for the confirmed
-- decimal-literal-division precision loss this avoids.
CAST(date_diff('second', l.disbursement_ts, m.principal_cure_ts) AS DOUBLE) / 3600.0 AS hours_to_principal_cure,
m.recovery_6h,
m.recovery_12h,
m.recovery_24h,
m.recovery_26h,
-- KNOWN LIMITATION (not fixed -- these don't reach the final borrower
-- output today): none of the checkpoint/outstanding fields below are
-- capped at principal, so an overpaid loan can produce pct_recovered_24h
-- > 1, negative unpaid_amount_24h/remaining_after_24h, and negative
-- outstanding_balance. If any of these intermediate fields are ever
-- selected into the final output or reused elsewhere, derive a
-- principal-allocation measure capped to [0, disbursed_amount] first
-- rather than using the raw cash totals directly.
m.recovery_24h / NULLIF(l.disbursed_amount,0) AS pct_recovered_24h,
(l.disbursed_amount - m.recovery_24h) AS unpaid_amount_24h,
(l.disbursed_amount - m.recovery_24h) / NULLIF(l.disbursed_amount,0) AS pct_unpaid_24h,
(l.disbursed_amount - m.recovery_26h) AS unpaid_amount_26h,
(l.disbursed_amount - m.recovery_26h) / NULLIF(l.disbursed_amount,0) AS pct_unpaid_26h,
-- KNOWN LIMITATION (in-scope decision, not an oversight): for loans that
-- were EVER ANOMALY_OPEN, total_recovered/total_cure_cashflow are corrected
-- to loan_state_daily's authoritative lifetime_repaid_ugx (see above), but
-- on_time_24h_flag/on_time_26h_flag/default_24h_flag/default_26h_flag/
-- severe_default_48h_flag below are NOT corrected. Those flags need the
-- precise TIMESTAMP each cure milestone was reached, itself derived
-- per-transaction from the same phonenumber-window attribution heuristic
-- responsible for the misattribution in the first place -- unlike the
-- scalar total, there is no equivalently authoritative replacement VALUE
-- for WHEN a transaction should have been attributed. This is a bounded,
-- accepted limitation. A real fix would need per-transaction requestid
-- matching (a separate, out-of-scope problem -- see the repayment_uid
-- cross-customer-collision work in data/repayment_uid_*.sql).
-- Maturity-gated: NULL until the loan has actually had this many hours to
-- cure, so an unseasoned loan reads as "not yet observable" rather than a
-- default. See the snapshot_ts comment in the snapshots CTE above.
CASE
WHEN (SELECT snapshot_ts FROM snapshots) < l.disbursement_ts + INTERVAL '24' HOUR THEN NULL
WHEN m.principal_cure_ts <= l.disbursement_ts + INTERVAL '24' HOUR THEN 1 ELSE 0
END AS on_time_24h_flag,
CASE
WHEN (SELECT snapshot_ts FROM snapshots) < l.disbursement_ts + INTERVAL '26' HOUR THEN NULL
WHEN m.principal_cure_ts <= l.disbursement_ts + INTERVAL '26' HOUR THEN 1 ELSE 0
END AS on_time_26h_flag,
CASE
WHEN (SELECT snapshot_ts FROM snapshots) < l.disbursement_ts + INTERVAL '26' HOUR THEN NULL
WHEN m.principal_cure_ts IS NULL OR m.principal_cure_ts > l.disbursement_ts + INTERVAL '26' HOUR THEN 1 ELSE 0
END AS default_26h_flag,
CASE
WHEN total_cure_cashflow = 0 THEN 'no_recovery'
WHEN m.principal_cure_ts IS NULL THEN 'partial_only'
WHEN m.principal_cure_ts <= l.disbursement_ts + INTERVAL '24' HOUR THEN 'on_time_24h'
WHEN m.principal_cure_ts <= l.disbursement_ts + INTERVAL '26' HOUR THEN 'grace_24_26h'
ELSE 'late_cured'
END AS repayment_bucket,
-- NEW FEATURES
m.recovery_6h / NULLIF(l.disbursed_amount,0) AS pct_recovered_6h,
m.recovery_12h / NULLIF(l.disbursed_amount,0) AS pct_recovered_12h,
(l.disbursed_amount - m.recovery_24h) AS remaining_after_24h,
(total_cure_cashflow - m.recovery_24h) AS recovery_after_24h,
-- Maturity-gated -- see on_time_24h_flag above.
CASE
WHEN (SELECT snapshot_ts FROM snapshots) < l.disbursement_ts + INTERVAL '24' HOUR THEN NULL
WHEN m.recovery_24h < l.disbursed_amount THEN 1 ELSE 0
END AS default_24h_flag,
CASE
WHEN (SELECT snapshot_ts FROM snapshots) < l.disbursement_ts + INTERVAL '48' HOUR THEN NULL
WHEN m.principal_cure_ts IS NULL
OR m.principal_cure_ts > l.disbursement_ts + INTERVAL '48' HOUR
THEN 1 ELSE 0
END AS severe_default_48h_flag,
-- Maturity-gated at the same 24h horizon as on_time_24h_flag/default_24h_flag
-- -- a loan disbursed shortly before snapshot_ts has trivially recovered $0
-- so far; without this gate lifetime_zero_recovery_rate would deteriorate
-- for a borrower purely because their latest loan hasn't had time to pay
-- yet, not because of any real non-payment behavior.
CASE
WHEN (SELECT snapshot_ts FROM snapshots) < l.disbursement_ts + INTERVAL '24' HOUR THEN NULL
WHEN total_cure_cashflow = 0 THEN 1 ELSE 0
END AS zero_recovery_flag,
l.first_recovery_ts,
l.last_recovery_ts,
date(l.first_recovery_ts) AS first_recovery_date,
date(l.last_recovery_ts) AS last_recovery_date,
date_diff('day', l.disbursement_ts, l.last_recovery_ts) AS days_to_last_recovery,
date_diff('hour', l.disbursement_ts, l.first_recovery_ts) AS hours_to_first_recovery,
date_diff('hour', l.disbursement_ts, l.last_recovery_ts) AS hours_to_last_recovery,
hour(l.disbursement_ts) AS disbursement_hour,
CASE
WHEN hour(l.disbursement_ts) BETWEEN 6 AND 11 THEN 'morning'
WHEN hour(l.disbursement_ts) BETWEEN 12 AND 17 THEN 'afternoon'
WHEN hour(l.disbursement_ts) BETWEEN 18 AND 22 THEN 'evening'
ELSE 'night'
END AS disbursement_time_bucket,
CASE
WHEN day_of_week(l.disbursement_ts) IN (6,7) THEN 'weekend'
ELSE 'weekday'
END AS disbursement_day_type,
-- OUTSTANDING & STATUS
(l.disbursed_amount - total_cure_cashflow) AS outstanding_balance,
CASE
WHEN total_cure_cashflow = 0 THEN 'no_recovery'
WHEN total_cure_cashflow < l.disbursed_amount THEN 'partial'
WHEN total_cure_cashflow >= l.disbursed_amount
AND l.total_penalty = 0 THEN 'closed'
WHEN total_cure_cashflow >= l.disbursed_amount
AND l.total_penalty > 0
AND total_cure_cashflow < l.disbursed_amount + l.total_penalty
THEN 'principal_cured_penalty_outstanding'
WHEN total_cure_cashflow >= l.disbursed_amount + l.total_penalty
AND l.total_penalty > 0
THEN 'closed_with_penalty'
ELSE 'unknown'
END AS loan_status,
-- Authoritative loan_state_daily read, attached here (per-loan grain, right
-- after loan_final) rather than at the very end after the borrower-level
-- window-function cascade -- these ls_* columns are pure pass-through for
-- everything between here and the final output (nothing in
-- borrower_history_base/streaks/exposure_features reads or filters on them),
-- EXCEPT ls_lifetime_repaid_ugx, which this same CTE (loan_final) already
-- consumes above to correct total_recovered/total_cure_cashflow for
-- ever-ANOMALY_OPEN loans -- so joining earlier vs. later produces identical
-- final values per row, keyed only by requestid = disbursement_fid either
-- way. Joining here lets the ##BORROWER_HISTORY_CHECKPOINT_1##
-- materialization below carry these columns forward without a second read
-- of loan_state_snapshot downstream.
--
-- KNOWN LIMITATION (confirmed via data/spot_check_loan_trace.sql, not a
-- guess): when a borrower takes a second disbursement on the SAME calendar
-- day as an already-open loan, that second disbursement's own
-- disbursement_fid gets no independent loan_state_daily row at all -- its
-- principal appears to be folded into the first same-day loan's ongoing
-- state instead (observed: two 750,000 disbursements 22h apart on the same
-- date, one showing loan_state_daily's lifetime_disbursed_ugx = 1,500,000
-- from day one, the other absent from loan_state_daily entirely). This
-- LEFT JOIN handles that safely -- the second loan's ls_* columns just come
-- back NULL, no error -- but expect NULL ls_loan_status/ls_aging_bucket/etc.
-- for any loan that isn't the first disbursement a borrower took that day.
-- Prevalence across the full loan book is not yet quantified.
COALESCE(eao.ever_anomaly_open, false) AS ever_anomaly_open,
ls.lifetime_repaid_ugx AS ls_lifetime_repaid_ugx,
ls.loan_status AS ls_loan_status,
ls.aging_bucket AS ls_aging_bucket,
ls.days_aging AS ls_days_aging,
ls.is_active_loan AS ls_is_active_loan,
ls.has_pre_window_history AS ls_has_pre_window_history,
-- KNOWN LIMITATION, CONFIRMED PERMANENT BY THE TABLE OWNER (not a data-
-- loading gap): interest_and_penalty_ugx/expected_total_charge_ugx/
-- charge_variance_ugx/charge_variance_pct are 100% NULL across every
-- loan_status value in analytics.momo_loan_book_tracker_loan_state_daily
-- for XtraFloat specifically -- confirmed via a live diagnostic
-- (data/interest_and_penalty_check.sql, data/expected_charge_check.sql,
-- data/charge_fields_all_status_check.sql) covering SETTLED, CLOSED, OPEN,
-- and ANOMALY_OPEN alike. latest_interest_and_penalty_ugx/latest_expected_
-- total_charge_ugx/latest_charge_variance_ugx/latest_charge_variance_pct
-- in the final borrower-level output will therefore always be NULL for
-- every row. Kept as output columns for schema consistency rather than
-- dropped -- do not treat their nullness as a bug to fix.
ls.interest_and_penalty_ugx AS ls_interest_and_penalty_ugx,
ls.expected_total_charge_ugx AS ls_expected_total_charge_ugx,
ls.charge_variance_ugx AS ls_charge_variance_ugx,
ls.charge_variance_pct AS ls_charge_variance_pct,
ls.is_charge_anomaly AS ls_is_charge_anomaly
FROM loan_core l
LEFT JOIN milestones m
ON l.requestid = m.requestid
AND l.phonenumber = m.phonenumber
LEFT JOIN loan_state_snapshot ls
ON l.requestid = ls.disbursement_fid
LEFT JOIN loan_ever_anomaly_open eao
ON l.requestid = eao.disbursement_fid
)
SELECT * FROM loan_final

WITH borrower_history_base AS (
SELECT
lf.*,
COUNT(*) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
) AS num_prior_loans,
AVG(on_time_24h_flag) OVER (

ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
) AS prior_on_time_24h_rate,

PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
) AS avg_prior_hours_to_cure,

PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
) AS worst_prior_hours_to_cure,
STDDEV(hours_to_principal_cure) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
) AS cure_time_volatility,
AVG(on_time_24h_flag) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN 3 PRECEDING AND 1 PRECEDING
) AS recent_3_on_time_rate,
AVG(hours_to_principal_cure) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN 3 PRECEDING AND 1 PRECEDING
) AS recent_3_avg_cure_time,
-- NEW DEFAULT RATES
-- average default rate across all prior loans only
AVG(default_24h_flag) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
) AS lifetime_prior_default_24h_rate,
AVG(default_24h_flag) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN 5 PRECEDING AND 1 PRECEDING
) AS recent_5_default_24h_rate,

(
AVG(hours_to_principal_cure) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN 3 PRECEDING AND 1 PRECEDING
)
-
AVG(hours_to_principal_cure) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN 6 PRECEDING AND 4 PRECEDING
)
) AS cure_time_trend
FROM {{CHECKPOINT_TABLE_1}} lf
),
borrower_history AS (

bhb.*,
CASE
WHEN cure_time_trend IS NULL THEN 'insufficient_history'
WHEN cure_time_trend < 0 THEN 'improving'
WHEN cure_time_trend > 0 THEN 'deteriorating'
ELSE 'stable'
END AS borrower_trend
FROM borrower_history_base bhb
),

WHEN num_prior_loans < 3 THEN 'thin_file'
WHEN cure_time_volatility IS NULL THEN 'insufficient_history'
ELSE 'thick_file'
END AS borrower_profile_type
FROM borrower_history bh
),

recent_loan_windows AS (
SELECT
bhf.*,
-- rolling loan count (proxy for 30d)
COUNT(*) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN 50 PRECEDING AND 1 PRECEDING
) AS loans_last_50_loans,
-- rolling defaults
COALESCE(
SUM(default_24h_flag) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN 50 PRECEDING AND 1 PRECEDING
),
0
) AS defaults_last_50_loans,
-- COUNT() of a nullable column (not COUNT(*)) skips NULLs, so this is the
-- number of prior loans in the window with a KNOWN 24h outcome -- the
-- correct denominator for default_rate_last_50_loans in `streaks` below.
-- loans_last_50_loans counts ALL prior loans (an activity measure) and
-- would understate the rate if divided into defaults_last_50_loans, which
-- (via SUM's own NULL-skipping) only counts mature ones.
COUNT(default_24h_flag) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN 50 PRECEDING AND 1 PRECEDING
) AS mature_loans_last_50,
-- streak reset groups
SUM(CASE WHEN on_time_24h_flag = 0 THEN 1 ELSE 0 END) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
) AS on_time_streak_group,
SUM(CASE WHEN default_24h_flag = 0 THEN 1 ELSE 0 END) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
) AS default_streak_group
FROM borrower_history_final bhf
),
streaks AS (
SELECT
rlw.*,

-- known 24h outcome), not loans_last_50_loans (all loans including
-- unseasoned ones), so an immature tail loan can't dilute the rate.
CASE
WHEN mature_loans_last_50 >= 5
THEN defaults_last_50_loans * 1.0 / mature_loans_last_50
ELSE NULL
END AS default_rate_last_50_loans,
-- streak counters
SUM(CASE WHEN on_time_24h_flag = 1 THEN 1 ELSE 0 END) OVER (
PARTITION BY phonenumber, on_time_streak_group
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW

SUM(CASE WHEN default_24h_flag = 1 THEN 1 ELSE 0 END) OVER (
PARTITION BY phonenumber, default_streak_group
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
) AS default_streak_current
FROM recent_loan_windows rlw
),
streaks_final AS (
SELECT
s.*,
COALESCE(LAG(on_time_streak_current) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
), 0) AS prior_on_time_streak,
COALESCE(LAG(default_streak_current) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
), 0) AS prior_default_streak
FROM streaks s
),
exposure_features AS (
SELECT
bh.*,
AVG(disbursed_amount) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
) AS avg_prior_loan_size,
-- loan_size_vs_history_ratio (an identical duplicate of loan_size_vs_
-- avg_ratio below, per the original "kept both, since you had both"

MAX(disbursed_amount) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
) AS max_prior_loan_size,
-- RELATIVE SIZE (kept both, since you had both)
disbursed_amount / NULLIF(
AVG(disbursed_amount) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
), 0
) AS loan_size_vs_avg_ratio,
disbursed_amount / NULLIF(
MAX(disbursed_amount) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
), 0
) AS loan_size_vs_max_ratio,
-- alternate logic = increase vs prior loan. did limit increase versus last loan?
CASE
WHEN num_prior_loans = 0 THEN 0
WHEN disbursed_amount > MAX(disbursed_amount) OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts, requestid
ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
) THEN 1
ELSE 0
END AS loan_above_prior_max_flag
FROM streaks_final bh

-- ============================================================================
-- Bucket 1 — Lifetime aggregates: one row per phonenumber, computed across
-- ALL of the borrower's loans via GROUP BY -- including the outcome of the
-- borrower's most recent (latest) loan.
--
-- INTENDED USE / LEAKAGE CAVEAT: this output feeds
-- prepare_borrower_limit_features() -> the caps/tier credit-LIMIT engine
-- (extrafloat/engine/run_extrafloat_limit_engine.py), which makes a
-- forward-looking decision -- "what limit for this borrower's NEXT loan" --
-- using everything currently known about them. Including the latest loan's
-- already-resolved outcome in these lifetime rates is correct for that use:
-- a live risk snapshot should reflect the most recent completed loan, not
-- pretend it hasn't happened yet.
-- This is NOT the same table the PD (probability-of-default) model trains
-- on -- that model uses its own loan-grain source
-- (data/loan_state_query_updated_materialized.txt, one row per target loan)
-- with its own dedicated label-eligibility/leakage-guard machinery (see
-- pd_model/preprocessing/loan_history_features.py). Do NOT repurpose this
-- borrower-level table to backtest or train a model whose target is the
-- latest loan's own outcome -- lifetime_on_time_24h_rate etc. already
-- contain that outcome, which would be leakage in that context even though
-- it is by-design correct for live limit-engine scoring.
-- ============================================================================
WITH borrower_lifetime_aggregates AS (
SELECT
phonenumber,
COUNT(*) AS total_loans,
MIN(disbursement_ts) AS first_loan_ts,
MAX(disbursement_ts) AS latest_loan_ts,
SUM(disbursed_amount) AS total_disbursed_amount,
AVG(disbursed_amount) AS avg_loan_size_lifetime,
MAX(disbursed_amount) AS max_loan_size_lifetime,
AVG(on_time_24h_flag) AS lifetime_on_time_24h_rate,
AVG(on_time_26h_flag) AS lifetime_on_time_26h_rate,
AVG(default_24h_flag) AS lifetime_default_24h_rate,
AVG(default_26h_flag) AS lifetime_default_26h_rate,
AVG(severe_default_48h_flag) AS lifetime_severe_default_48h_rate,
AVG(zero_recovery_flag) AS lifetime_zero_recovery_rate,
AVG(hours_to_principal_cure) AS lifetime_avg_hours_to_principal_cure,
MAX(hours_to_principal_cure) AS lifetime_worst_hours_to_principal_cure,
STDDEV(hours_to_principal_cure) AS lifetime_cure_time_volatility,
-- Borrower already had a loan before the data feed started (2026-01-01) on
-- ANY of their loans -- true if this borrower's own lifetime stats above
-- are left-censored (undercounted) rather than reflecting their full history.
bool_or(ls_has_pre_window_history) AS has_pre_window_history
FROM {{CHECKPOINT_TABLE_2}}
GROUP BY phonenumber
),
-- ============================================================================
-- Bucket 2 — Latest-loan snapshot: the single most recent loan per borrower,
-- selected the same way prepare_borrower_limit_features() deduplicates
-- (latest_disbursement_ts desc, latest_requestid desc — see
-- extrafloat_limit_engine_features.py lines 317-326).
-- ============================================================================
latest_loan_snapshot AS (
SELECT
phonenumber,
requestid AS latest_requestid,
disbursement_ts AS latest_disbursement_ts,
disbursed_amount AS latest_disbursed_amount,
num_prior_loans,
prior_on_time_24h_rate,
avg_prior_hours_to_cure,
worst_prior_hours_to_cure,
cure_time_volatility,
recent_3_on_time_rate,
recent_3_avg_cure_time,
lifetime_prior_default_24h_rate,
recent_5_default_24h_rate,
cure_time_trend,
borrower_trend,
borrower_profile_type,
loans_last_50_loans,
defaults_last_50_loans,
default_rate_last_50_loans,
prior_on_time_streak,
prior_default_streak,
avg_prior_loan_size,
max_prior_loan_size,
loan_size_vs_avg_ratio,
loan_size_vs_max_ratio,
loan_above_prior_max_flag,
-- Authoritative loan_state_daily read for this borrower's latest loan.
ls_loan_status AS latest_loan_status,
ls_aging_bucket AS latest_aging_bucket,
ls_days_aging AS latest_days_aging,
ls_is_active_loan AS latest_is_active_loan,
ls_interest_and_penalty_ugx AS latest_interest_and_penalty_ugx,
ls_expected_total_charge_ugx AS latest_expected_total_charge_ugx,
ls_charge_variance_ugx AS latest_charge_variance_ugx,
ls_charge_variance_pct AS latest_charge_variance_pct,
ls_is_charge_anomaly AS latest_is_charge_anomaly,
ROW_NUMBER() OVER (
PARTITION BY phonenumber
ORDER BY disbursement_ts DESC, requestid DESC
) AS rn
FROM {{CHECKPOINT_TABLE_2}}
)
-- ============================================================================
-- Final projection — borrower-level dataset. The original 41 columns keep
-- their exact names/order (still matching load_borrower_limit_features());
-- new columns are appended at the end.
-- ============================================================================
SELECT
a.phonenumber,
a.total_loans,
a.first_loan_ts,
a.latest_loan_ts,
a.total_disbursed_amount,
a.avg_loan_size_lifetime,
a.max_loan_size_lifetime,
a.lifetime_on_time_24h_rate,
a.lifetime_on_time_26h_rate,
a.lifetime_default_24h_rate,
a.lifetime_default_26h_rate,
a.lifetime_severe_default_48h_rate,
a.lifetime_zero_recovery_rate,
a.lifetime_avg_hours_to_principal_cure,
a.lifetime_worst_hours_to_principal_cure,
a.lifetime_cure_time_volatility,
b.latest_requestid,
b.latest_disbursement_ts,
b.latest_disbursed_amount,
b.num_prior_loans,
b.prior_on_time_24h_rate,
b.avg_prior_hours_to_cure,
b.worst_prior_hours_to_cure,
b.cure_time_volatility,
b.recent_3_on_time_rate,
b.recent_3_avg_cure_time,
b.lifetime_prior_default_24h_rate,
b.recent_5_default_24h_rate,
b.cure_time_trend,
b.borrower_trend,
b.borrower_profile_type,
b.loans_last_50_loans,
b.defaults_last_50_loans,
b.default_rate_last_50_loans,
b.prior_on_time_streak,
b.prior_default_streak,
b.avg_prior_loan_size,
b.max_prior_loan_size,
b.loan_size_vs_avg_ratio,
b.loan_size_vs_max_ratio,
b.loan_above_prior_max_flag,
-- New columns (additive) --
a.has_pre_window_history,
b.latest_loan_status,
b.latest_aging_bucket,
b.latest_days_aging,
b.latest_is_active_loan,
b.latest_interest_and_penalty_ugx,
b.latest_expected_total_charge_ugx,
b.latest_charge_variance_ugx,
b.latest_charge_variance_pct,
b.latest_is_charge_anomaly
FROM borrower_lifetime_aggregates a
INNER JOIN (
SELECT * FROM latest_loan_snapshot WHERE rn = 1
) b
ON a.phonenumber = b.phonenumber
