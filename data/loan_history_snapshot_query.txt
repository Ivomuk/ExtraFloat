-- =====================================================================
-- ExtraFloat loan history -- SNAPSHOT VERSION for the credit limit engine
-- SQL dialect: Trino
--
-- WHAT THIS FILE IS
--   A snapshot-style sibling to loan_state_query_updated.txt, shaped
--   like data/loan_summary_query.txt:
--
--     - ONE snapshot_dt for the whole population (edit the `snapshots`
--       CTE below), not a per-loan disbursement date.
--     - One row per msisdn "as of" that date, not one row per
--       disbursement_fid.
--     - No `split` column -- there is no train/validation cohort here.
--     - No bad_state_* labels, no future-outcome columns -- there is no
--       forward-looking window at a snapshot date, so nothing in this
--       file requires waiting for future loan_state_daily rows.
--
--   Everything else -- point-in-time discipline, the per-loan
--   lifetime_* semantics, the prior-loan history aggregates -- reuses
--   the same logic as loan_state_query_updated.txt.
--
-- DEPENDENCY
--   Reads hive.analytics.tmp_disbursements and hive.analytics.tmp_loan_state,
--   the two base tables built by statements 1-2 of
--   loan_state_query_updated_materialized.txt. Run those two CREATE TABLE
--   statements first (or refresh them) so they cover data through the
--   snapshot_dt below. tmp_target_loans is NOT needed here -- that table
--   exists only to resolve disbursement_fid -> loan_uid for a *new*
--   target loan being labeled, and there is no new loan being disbursed
--   at a snapshot date, so the same-day bridge/mapping machinery does
--   not apply.
--
-- POINT-IN-TIME DISCIPLINE
--   loan_state_query_updated.txt uses `state_date < loan_date` (strict)
--   for its scoring-time features, because it is asking "what did we
--   know right before this specific loan's exact disbursement instant."
--   This file asks a different question -- "what do we know as of the
--   end of snapshot_dt" -- so it uses `<=` (inclusive) throughout,
--   matching the convention in loan_summary_query.txt.
--
-- NOT A DROP-IN REPLACEMENT FOR ANY EXISTING ENGINE INPUT FILE
--   The credit limit engine currently takes three files: transaction
--   capacity (data/transaction_capacity_features_sample.txt), borrower
--   history (data/borrower_history.txt), and loan summary
--   (data/loan_summary_query.txt). This file replaces neither.
--
--   - data/loan_summary_query.txt feeds compute_recent_usage_cap(),
--     which requires disbursement_vol/val_1m, repayment_vol/val_1m,
--     penalties_1m, and their 3m variants -- trailing CASH-FLOW volume,
--     sourced from event-level devdata.xtrafloat_daily_trans rows. This
--     file cannot produce those: momo_loan_book_tracker_loan_state_daily
--     only exposes cumulative per-loan running totals as of each daily
--     snapshot, not a repayment/penalty event log, so there is no way to
--     derive a trailing-30-day repayment or penalty FLOW from it without
--     either differencing two point-in-time snapshots or pulling in
--     xtrafloat_daily_trans as well (which would make this file a mixed-
--     source query, not a clean sibling to loan_state_query_updated.txt).
--
--   - data/borrower_history.txt (== borrower_credit_limit_aggregated.txt,
--     feeding load_borrower_limit_features() / prepare_borrower_limit_
--     features()) expects hour-level cure-time fields
--     (lifetime_on_time_24h_rate, hours_to_principal_cure, recovery_24h,
--     ...) reconstructed from the same raw devdata.xtrafloat_daily_trans
--     transactions. momo_loan_book_tracker_loan_state_daily only has
--     day-level granularity (days_aging, tenure_days), so there is no
--     hour-level cure data to compute those fields from here either.
--
--   What this file DOES add that is not already produced anywhere else
--   in the pipeline: a validated, point-in-time-correct view of an
--   agent's CURRENT loan state as of an arbitrary snapshot date --
--   has_unresolved_loan_at_snapshot, active_loan_days_aging_at_snapshot,
--   active_loan_outstanding_ugx_at_snapshot, anomaly_open_at_snapshot --
--   plus historical_anomaly_open_loan_rate using the corrected per-loan
--   lifetime_* semantics documented in loan_state_query_updated.txt. None
--   of the three existing engine input files carry a live "is this agent
--   currently delinquent or mid-rollover, and by how much" signal.
--
--   This file is complementary: a fourth, additive input, not a
--   replacement for any of the three above. Whether and how to wire it
--   into the engine (a new loader, a new cap, or folding specific
--   columns into an existing cap's inputs) is a separate decision.
-- =====================================================================

WITH

-- ---------------------------------------------------------------------
-- 1. Snapshot date. Edit this to move the "as of" point -- must be <=
-- the base tables' actual max state_date (re-check via
-- SELECT MAX(state_date) FROM hive.analytics.tmp_loan_state, since
-- source data availability moves forward over time). Last confirmed:
-- 2026-08-17 (SELECT MAX(state_date) FROM hive.analytics.tmp_loan_state).
-- ---------------------------------------------------------------------
snapshots AS (
    SELECT CAST(20260817 AS BIGINT) AS snapshot_dt
),

-- ---------------------------------------------------------------------
-- 2. Base population: every agent who had disbursed at least one loan
--    on or before snapshot_dt. Excludes agents whose first loan is
--    still in the future relative to the snapshot (no look-ahead).
-- ---------------------------------------------------------------------
base AS (
    SELECT DISTINCT
        d.msisdn,
        s.snapshot_dt,
        CAST(
            DATE_PARSE(CAST(s.snapshot_dt AS VARCHAR), '%Y%m%d')
            AS DATE
        ) AS snapshot_date
    FROM hive.analytics.tmp_disbursements d
    CROSS JOIN snapshots s
    WHERE d.loan_date <= CAST(
        DATE_PARSE(CAST(s.snapshot_dt AS VARCHAR), '%Y%m%d')
        AS DATE
    )
),

-- ---------------------------------------------------------------------
-- 3. One row per (msisdn, loan_uid): the most recent known state of
--    that loan on or before snapshot_dt. Unlike
--    prior_loan_state_candidates in the training query, nothing is
--    excluded here -- there is no target loan_uid to keep separate,
--    so this is simply every loan the agent has ever had, as it stood
--    at snapshot time.
-- ---------------------------------------------------------------------
agent_loan_state_candidates AS (
    SELECT
        b.msisdn,
        b.snapshot_date,

        ls.loan_uid,
        ls.loan_seq,
        ls.current_loan_start_date,
        ls.closure_date,
        ls.tenure_days,
        ls.aging_bucket,
        ls.loan_status,
        ls.is_anomaly_open,
        ls.is_active_loan,
        ls.has_pre_window_history,
        ls.days_aging,
        ls.outstanding_ugx,
        ls.last_repayment_date,
        ls.state_date,

        ls.loan_disbursed_ugx,
        ls.loan_repaid_ugx,
        ls.loan_gross_repaid_ugx,
        ls.loan_fees_paid_ugx,
        ls.loan_interest_paid_ugx,

        ls.disbursement_event_count,
        ls.repayment_event_count,
        ls.fee_event_count,
        ls.interest_event_count,

        ROW_NUMBER() OVER (
            PARTITION BY b.msisdn, ls.loan_uid
            ORDER BY
                ls.state_date DESC,
                ls.inserted_ts DESC
        ) AS loan_state_rn

    FROM base b

    JOIN hive.analytics.tmp_loan_state ls
      ON ls.customer_msisdn = b.msisdn
     AND ls.loan_status <> 'NEVER_BORROWED'
     AND ls.current_loan_start_date <= b.snapshot_date
     AND ls.state_date <= b.snapshot_date
),

agent_loans AS (
    SELECT *
    FROM agent_loan_state_candidates
    WHERE loan_state_rn = 1
),

-- Rank every known loan by recency (most recent current_loan_start_date
-- first). Position 1 is "the loan as it stands at snapshot time" --
-- whether still open or already closed.
loans_ranked AS (
    SELECT
        a.*,

        ROW_NUMBER() OVER (
            PARTITION BY a.msisdn
            ORDER BY
                a.current_loan_start_date DESC,
                a.loan_seq DESC,
                a.loan_uid DESC
        ) AS loan_recency

    FROM agent_loans a
),

latest_state AS (
    SELECT *
    FROM loans_ranked
    WHERE loan_recency = 1
),

-- Closed loans get a separate ranking, exactly as in
-- loan_state_query_updated.txt, so max_tenure_days_last_5_closed_loans
-- and the on-time streak walk back through closed loans only.
-- loan_status IN (...) is required alongside closure_date IS NOT NULL --
-- confirmed via a direct data check that the source table populates
-- closure_date on genuinely non-terminal rows too (e.g. ANOMALY_OPEN
-- rows carrying a stale/unrelated date while the loan was still open),
-- so closure_date alone is not reliable evidence a loan actually closed
-- (same finding and fix as loan_state_query_updated_materialized.txt's
-- closed_prior_loans_ranked).
closed_loans_ranked AS (
    SELECT
        a.*,

        ROW_NUMBER() OVER (
            PARTITION BY a.msisdn
            ORDER BY
                a.closure_date DESC,
                a.current_loan_start_date DESC,
                a.loan_seq DESC,
                a.loan_uid DESC
        ) AS closed_loan_recency

    FROM agent_loans a
    WHERE a.closure_date IS NOT NULL
      AND a.loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
),

-- ---------------------------------------------------------------------
-- 4. As-of-snapshot state features (mirrors scoring_state_features,
--    anchored at snapshot_date instead of a target loan's own date).
-- ---------------------------------------------------------------------
latest_state_features AS (
    SELECT
        b.msisdn,

        CASE WHEN ls.msisdn IS NULL THEN 0 ELSE 1 END
            AS has_observed_prior_state,

        COALESCE(
            CASE WHEN ls.has_pre_window_history THEN 1 ELSE 0 END,
            0
        ) AS has_pre_window_history,

        CASE
            WHEN COALESCE(ls.is_active_loan, FALSE)
              OR ls.loan_status IN (
                    'OPEN',
                    'OPEN_PRE_WINDOW',
                    'ANOMALY_OPEN'
                 )
            THEN 1
            ELSE 0
        END AS has_unresolved_loan_at_snapshot,

        CASE
            WHEN COALESCE(ls.is_active_loan, FALSE)
              OR ls.loan_status IN (
                    'OPEN',
                    'OPEN_PRE_WINDOW',
                    'ANOMALY_OPEN'
                 )
            THEN COALESCE(ls.days_aging, 0)
            ELSE 0
        END AS active_loan_days_aging_at_snapshot,

        CASE
            WHEN COALESCE(ls.is_active_loan, FALSE)
              OR ls.loan_status IN (
                    'OPEN',
                    'OPEN_PRE_WINDOW',
                    'ANOMALY_OPEN'
                 )
            THEN COALESCE(ls.outstanding_ugx, 0)
            ELSE 0
        END AS active_loan_outstanding_ugx_at_snapshot,

        CASE
            WHEN COALESCE(ls.is_active_loan, FALSE)
              OR ls.loan_status IN (
                    'OPEN',
                    'OPEN_PRE_WINDOW',
                    'ANOMALY_OPEN'
                 )
            THEN ls.aging_bucket
        END AS active_loan_aging_bucket_at_snapshot,

        CASE
            WHEN ls.loan_status = 'ANOMALY_OPEN'
              OR COALESCE(ls.is_anomaly_open, FALSE)
            THEN 1
            ELSE 0
        END AS anomaly_open_at_snapshot,

        ls.state_date AS latest_state_date,
        ls.loan_uid AS latest_loan_uid,
        ls.loan_seq AS loan_seq_at_snapshot,
        ls.loan_status AS latest_loan_status,

        CASE
            WHEN ls.last_repayment_date IS NOT NULL
            THEN DATE_DIFF(
                'day',
                ls.last_repayment_date,
                b.snapshot_date
            )
        END AS days_since_last_repayment,

        ls.loan_disbursed_ugx AS latest_loan_disbursed_ugx,
        ls.loan_repaid_ugx AS latest_loan_repaid_ugx,
        ls.loan_gross_repaid_ugx AS latest_loan_gross_repaid_ugx,
        ls.loan_fees_paid_ugx AS latest_loan_fees_paid_ugx,
        ls.loan_interest_paid_ugx AS latest_loan_interest_paid_ugx,

        CASE
            WHEN ls.loan_disbursed_ugx > 0
            THEN ls.loan_repaid_ugx / ls.loan_disbursed_ugx
        END AS latest_loan_principal_repayment_ratio,

        CASE
            WHEN ls.loan_disbursed_ugx > 0
            THEN ls.loan_gross_repaid_ugx / ls.loan_disbursed_ugx
        END AS latest_loan_gross_repayment_ratio,

        CASE
            WHEN ls.disbursement_event_count > 0
            THEN
                TRY_CAST(ls.repayment_event_count AS DOUBLE)
                / TRY_CAST(ls.disbursement_event_count AS DOUBLE)
        END AS latest_loan_repayment_to_disbursement_event_ratio

    FROM base b

    LEFT JOIN latest_state ls
      ON ls.msisdn = b.msisdn
),

-- ---------------------------------------------------------------------
-- 5. History across every known loan as of snapshot_dt (mirrors
--    prior_loan_features -- here "prior" includes whatever loan is
--    currently active, since there is no separate target loan to hold
--    apart from the rest of the history).
-- ---------------------------------------------------------------------
loan_history_features AS (
    SELECT
        msisdn,

        COUNT(*) AS observed_loan_count,

        -- loan_status IN (...) required alongside closure_date IS NOT
        -- NULL in every "closed loan" condition below -- see
        -- closed_loans_ranked's comment above for why.
        COUNT_IF(
            closure_date IS NOT NULL
            AND loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
        ) AS closed_loan_count,

        AVG(
            CASE
                WHEN closure_date IS NOT NULL
                 AND loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
                THEN TRY_CAST(tenure_days AS DOUBLE)
            END
        ) AS avg_tenure_days_closed_loans,

        SUM(
            CASE
                WHEN loan_status = 'ANOMALY_OPEN'
                  OR COALESCE(is_anomaly_open, FALSE)
                THEN 1
                ELSE 0
            END
        ) AS historical_anomaly_open_loan_count,

        AVG(
            CASE
                WHEN loan_status = 'ANOMALY_OPEN'
                  OR COALESCE(is_anomaly_open, FALSE)
                THEN 1.0
                ELSE 0.0
            END
        ) AS historical_anomaly_open_loan_rate,

        SUM(COALESCE(loan_disbursed_ugx, 0))
            AS all_loans_disbursed_ugx,

        SUM(COALESCE(loan_repaid_ugx, 0))
            AS all_loans_repaid_ugx,

        SUM(COALESCE(loan_gross_repaid_ugx, 0))
            AS all_loans_gross_repaid_ugx,

        SUM(COALESCE(loan_fees_paid_ugx, 0))
            AS all_loans_fees_paid_ugx,

        SUM(COALESCE(loan_interest_paid_ugx, 0))
            AS all_loans_interest_paid_ugx,

        CASE
            WHEN SUM(COALESCE(loan_disbursed_ugx, 0)) > 0
            THEN
                SUM(COALESCE(loan_repaid_ugx, 0))
                / SUM(COALESCE(loan_disbursed_ugx, 0))
        END AS all_loans_principal_repayment_ratio,

        CASE
            WHEN SUM(COALESCE(loan_disbursed_ugx, 0)) > 0
            THEN
                SUM(COALESCE(loan_gross_repaid_ugx, 0))
                / SUM(COALESCE(loan_disbursed_ugx, 0))
        END AS all_loans_gross_repayment_ratio,

        AVG(
            CASE
                WHEN loan_disbursed_ugx > 0
                THEN loan_repaid_ugx / loan_disbursed_ugx
            END
        ) AS avg_loan_principal_repayment_ratio,

        -- Closed-loan-only ratio avoids mechanically penalizing an agent
        -- merely because a loan is still open at snapshot time.
        -- loan_status IN (...) required alongside closure_date IS NOT
        -- NULL -- see closed_loans_ranked's comment above for why.
        CASE
            WHEN SUM(
                CASE
                    WHEN closure_date IS NOT NULL
                     AND loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
                    THEN COALESCE(loan_disbursed_ugx, 0)
                    ELSE 0
                END
            ) > 0
            THEN
                SUM(
                    CASE
                        WHEN closure_date IS NOT NULL
                         AND loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
                        THEN COALESCE(loan_repaid_ugx, 0)
                        ELSE 0
                    END
                )
                /
                SUM(
                    CASE
                        WHEN closure_date IS NOT NULL
                         AND loan_status IN ('SETTLED', 'CLOSED', 'OVERPAID')
                        THEN COALESCE(loan_disbursed_ugx, 0)
                        ELSE 0
                    END
                )
        END AS closed_loans_principal_repayment_ratio

    FROM loans_ranked
    GROUP BY msisdn
),

closed_loan_features AS (
    SELECT
        msisdn,

        MAX(
            CASE
                WHEN closed_loan_recency <= 5
                THEN tenure_days
            END
        ) AS max_tenure_days_last_5_closed_loans

    FROM closed_loans_ranked
    GROUP BY msisdn
),

last_closed_loan_features AS (
    SELECT
        msisdn,

        loan_uid AS last_closed_loan_uid,
        closure_date AS last_closed_loan_closure_date,
        tenure_days AS last_closed_loan_tenure_days,
        aging_bucket AS last_closed_loan_aging_bucket,

        loan_disbursed_ugx AS last_closed_loan_disbursed_ugx,
        loan_repaid_ugx AS last_closed_loan_repaid_ugx,
        loan_gross_repaid_ugx AS last_closed_loan_gross_repaid_ugx,

        CASE
            WHEN loan_disbursed_ugx > 0
            THEN loan_repaid_ugx / loan_disbursed_ugx
        END AS last_closed_loan_principal_repayment_ratio

    FROM closed_loans_ranked
    WHERE closed_loan_recency = 1
),

-- tenure_days <= 1 is provisional -- see the same caveat in
-- loan_state_query_updated.txt.
closed_loan_streak_rows AS (
    SELECT
        msisdn,
        closed_loan_recency,
        tenure_days,

        SUM(
            CASE
                WHEN tenure_days > 1
                  OR tenure_days IS NULL
                THEN 1
                ELSE 0
            END
        ) OVER (
            PARTITION BY msisdn
            ORDER BY closed_loan_recency
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS late_loans_seen

    FROM closed_loans_ranked
),

on_time_streak AS (
    SELECT
        msisdn,

        SUM(
            CASE
                WHEN tenure_days <= 1
                 AND late_loans_seen = 0
                THEN 1
                ELSE 0
            END
        ) AS consecutive_on_time_loans

    FROM closed_loan_streak_rows
    GROUP BY msisdn
),

-- ---------------------------------------------------------------------
-- Late-payment penalty history, sourced from the authoritative penalty
-- ledger (analytics.momo_loan_book_tracker_xtrafloat_loan_state_daily,
-- a separate, XtraFloat-only table -- NOT
-- momo_loan_book_tracker_loan_state_daily used elsewhere in this file).
-- Identical to loan_state_query_updated_materialized.txt's CTEs of the
-- same name -- a loan's "was it ever penalized" summary doesn't depend
-- on which file consumes it. This is a daily-snapshot table (one row per
-- (disbursement_fid, date_key)) -- dedup to one row per (disbursement_fid,
-- date_key) first, then collapse to one row per disbursement_fid.
-- penalty_1_due/penalty_2_due are confirmed (via live warehouse checks)
-- to genuinely vary -- true once a loan misses the 24h/48h deadline and
-- is charged a real penalty, not a universal per-loan flag -- and to
-- stay true across a loan's later daily rows even after that specific
-- penalty amount is paid, so bool_or here correctly captures "was this
-- loan EVER penalized," not just its most recent day's state.
-- ---------------------------------------------------------------------
xtrafloat_penalty_loan_daily_ranked AS (
    SELECT
        pen.disbursement_fid,
        pen.date_key,
        pen.penalty_1_due,
        pen.penalty_1_owed_ugx,
        pen.penalty_2_due,
        pen.penalty_2_owed_ugx,

        ROW_NUMBER() OVER (
            PARTITION BY
                pen.disbursement_fid,
                pen.date_key
            ORDER BY
                pen.inserted_ts DESC
        ) AS penalty_day_rn

    FROM analytics.momo_loan_book_tracker_xtrafloat_loan_state_daily pen
    WHERE pen.ova = 'XTRAFLOAT-AGENT'
      AND pen.date_key >= 20260101
),

xtrafloat_penalty_loan_summary AS (
    SELECT
        disbursement_fid,

        bool_or(penalty_1_due) AS ever_penalty_1_due,
        bool_or(penalty_2_due) AS ever_penalty_2_due,
        MAX(penalty_1_owed_ugx) AS penalty_1_owed_ugx,
        MAX(penalty_2_owed_ugx) AS penalty_2_owed_ugx

    FROM xtrafloat_penalty_loan_daily_ranked
    WHERE penalty_day_rn = 1
    GROUP BY disbursement_fid
),

-- ---------------------------------------------------------------------
-- Late-payment penalty history, anchored at snapshot_date -- the
-- scoring-time mirror of loan_state_query_updated_materialized.txt's
-- prior_penalty_features. This file has no per-target-loan concept (one
-- row per msisdn "as of" snapshot_date), so this is keyed the same way
-- disbursement_history_features below is: every loan on or before
-- snapshot_date counts, not "prior to a specific target loan."
--
-- Named late_fee_*, not penalty_*, matching this file's own unprefixed
-- naming convention (vs. the training file's prior_* prefix -- see the
-- SNAPSHOT_TO_TRAINING_COLUMN_MAP mapping in
-- pd_model/preprocessing/loan_history_features.py) -- this also avoids
-- colliding with LEAKAGE_PATTERNS's blanket "penalt" substring block in
-- pd_model/config/feature_config.py, same reasoning as the training file.
-- ---------------------------------------------------------------------
penalty_history_features AS (
    SELECT
        b.msisdn,

        COUNT_IF(pen.ever_penalty_1_due)
            AS late_fee_1_count,

        COUNT_IF(pen.ever_penalty_2_due)
            AS late_fee_2_count,

        COALESCE(
            SUM(
                COALESCE(pen.penalty_1_owed_ugx, 0)
                + COALESCE(pen.penalty_2_owed_ugx, 0)
            ),
            0
        ) AS total_late_fee_owed_ugx

    FROM base b

    LEFT JOIN hive.analytics.tmp_disbursements pd
      ON pd.msisdn = b.msisdn
     AND pd.loan_date <= b.snapshot_date

    LEFT JOIN xtrafloat_penalty_loan_summary pen
      ON pen.disbursement_fid = pd.disbursement_fid

    GROUP BY b.msisdn
),

-- ---------------------------------------------------------------------
-- 6. Disbursement frequency and amount, anchored at snapshot_date
--    (mirrors prior_disbursement_features).
-- ---------------------------------------------------------------------
disbursement_history_features AS (
    SELECT
        b.msisdn,

        COUNT(pd.disbursement_fid) AS disbursement_count,

        COUNT_IF(
            pd.loan_date >= DATE_ADD('day', -30, b.snapshot_date)
        ) AS disbursement_count_30d,

        COUNT_IF(
            pd.loan_date >= DATE_ADD('day', -90, b.snapshot_date)
        ) AS disbursement_count_90d,

        COUNT_IF(
            pd.loan_date >= DATE_ADD('day', -180, b.snapshot_date)
        ) AS disbursement_count_180d,

        -- Thin-file routing signals, mirroring the training query's
        -- prior_loan_count_180d / prior_active_loan_months_180d.
        COUNT_IF(
            pd.loan_date >= DATE_ADD('day', -180, b.snapshot_date)
        ) AS prior_loan_count_180d,

        COUNT(
            DISTINCT CASE
                WHEN pd.loan_date >= DATE_ADD('day', -180, b.snapshot_date)
                THEN DATE_TRUNC('month', pd.loan_date)
            END
        ) AS prior_active_loan_months_180d,

        AVG(
            CASE
                WHEN pd.loan_date >= DATE_ADD('day', -30, b.snapshot_date)
                THEN pd.disbursement_amount_ugx
            END
        ) AS avg_loan_amount_30d,

        AVG(
            CASE
                WHEN pd.loan_date >= DATE_ADD('day', -90, b.snapshot_date)
                THEN pd.disbursement_amount_ugx
            END
        ) AS avg_loan_amount_90d,

        AVG(
            CASE
                WHEN pd.loan_date >= DATE_ADD('day', -180, b.snapshot_date)
                THEN pd.disbursement_amount_ugx
            END
        ) AS avg_loan_amount_180d,

        COALESCE(
            SUM(
                CASE
                    WHEN pd.loan_date
                        >= DATE_ADD('day', -30, b.snapshot_date)
                    THEN pd.disbursement_amount_ugx
                    ELSE 0
                END
            ),
            0
        ) AS disbursed_ugx_30d,

        COALESCE(
            SUM(
                CASE
                    WHEN pd.loan_date
                        >= DATE_ADD('day', -90, b.snapshot_date)
                    THEN pd.disbursement_amount_ugx
                    ELSE 0
                END
            ),
            0
        ) AS disbursed_ugx_90d,

        COALESCE(
            SUM(
                CASE
                    WHEN pd.loan_date
                        >= DATE_ADD('day', -180, b.snapshot_date)
                    THEN pd.disbursement_amount_ugx
                    ELSE 0
                END
            ),
            0
        ) AS disbursed_ugx_180d,

        -- b.snapshot_date is wrapped in MAX() here (constant per msisdn --
        -- base has exactly one row per msisdn, see the "1. Snapshot date"
        -- CTE above) purely so Trino accepts it inside this GROUP BY
        -- b.msisdn aggregate expression; every other reference to
        -- b.snapshot_date in this CTE is already inside a SUM(CASE ...)
        -- and needed no such wrapping.
        CASE
            WHEN COUNT(pd.disbursement_fid) > 0
            THEN
                DATE_DIFF(
                    'month',
                    MIN(pd.loan_date),
                    MAX(b.snapshot_date)
                ) + 1
        END AS months_since_first_observed_loan

    FROM base b

    LEFT JOIN hive.analytics.tmp_disbursements pd
      ON pd.msisdn = b.msisdn
     AND pd.loan_date <= b.snapshot_date

    GROUP BY b.msisdn
)

-- ---------------------------------------------------------------------
-- 7. Final snapshot dataset -- one row per msisdn.
-- ---------------------------------------------------------------------
SELECT
    b.msisdn,
    b.snapshot_dt,

    -- As-of-snapshot debt state
    COALESCE(lsf.has_observed_prior_state, 0)
        AS has_observed_prior_state,

    COALESCE(lsf.has_pre_window_history, 0)
        AS has_pre_window_history,

    COALESCE(lsf.has_unresolved_loan_at_snapshot, 0)
        AS has_unresolved_loan_at_snapshot,

    COALESCE(lsf.active_loan_days_aging_at_snapshot, 0)
        AS active_loan_days_aging_at_snapshot,

    COALESCE(lsf.active_loan_outstanding_ugx_at_snapshot, 0)
        AS active_loan_outstanding_ugx_at_snapshot,

    lsf.active_loan_aging_bucket_at_snapshot,

    COALESCE(lsf.anomaly_open_at_snapshot, 0)
        AS anomaly_open_at_snapshot,

    lsf.latest_state_date,
    lsf.latest_loan_uid,
    lsf.loan_seq_at_snapshot,
    lsf.latest_loan_status,
    lsf.days_since_last_repayment,

    lsf.latest_loan_disbursed_ugx,
    lsf.latest_loan_repaid_ugx,
    lsf.latest_loan_gross_repaid_ugx,
    lsf.latest_loan_fees_paid_ugx,
    lsf.latest_loan_interest_paid_ugx,
    lsf.latest_loan_principal_repayment_ratio,
    lsf.latest_loan_gross_repayment_ratio,
    lsf.latest_loan_repayment_to_disbursement_event_ratio,

    -- History across every known loan as of snapshot_dt
    COALESCE(h.observed_loan_count, 0)
        AS observed_loan_count,

    COALESCE(h.closed_loan_count, 0)
        AS closed_loan_count,

    CASE
        WHEN lsf.loan_seq_at_snapshot IS NOT NULL
        THEN
            lsf.loan_seq_at_snapshot
            - COALESCE(h.observed_loan_count, 0)
    END AS loan_seq_minus_observed_loan_count,

    h.avg_tenure_days_closed_loans,

    cl.max_tenure_days_last_5_closed_loans,

    COALESCE(h.historical_anomaly_open_loan_count, 0)
        AS historical_anomaly_open_loan_count,

    h.historical_anomaly_open_loan_rate,

    COALESCE(h.all_loans_disbursed_ugx, 0)
        AS all_loans_disbursed_ugx,

    COALESCE(h.all_loans_repaid_ugx, 0)
        AS all_loans_repaid_ugx,

    COALESCE(h.all_loans_gross_repaid_ugx, 0)
        AS all_loans_gross_repaid_ugx,

    COALESCE(h.all_loans_fees_paid_ugx, 0)
        AS all_loans_fees_paid_ugx,

    COALESCE(h.all_loans_interest_paid_ugx, 0)
        AS all_loans_interest_paid_ugx,

    h.all_loans_principal_repayment_ratio,
    h.all_loans_gross_repayment_ratio,
    h.closed_loans_principal_repayment_ratio,
    h.avg_loan_principal_repayment_ratio,

    -- Latest closed loan
    lc.last_closed_loan_uid,
    lc.last_closed_loan_closure_date,
    lc.last_closed_loan_tenure_days,
    lc.last_closed_loan_aging_bucket,
    lc.last_closed_loan_disbursed_ugx,
    lc.last_closed_loan_repaid_ugx,
    lc.last_closed_loan_gross_repaid_ugx,
    lc.last_closed_loan_principal_repayment_ratio,

    COALESCE(ots.consecutive_on_time_loans, 0)
        AS consecutive_on_time_loans,

    -- Disbursement frequency and amount
    COALESCE(dh.disbursement_count, 0)
        AS disbursement_count,

    COALESCE(dh.disbursement_count_30d, 0)
        AS disbursement_count_30d,

    COALESCE(dh.disbursement_count_90d, 0)
        AS disbursement_count_90d,

    COALESCE(dh.disbursement_count_180d, 0)
        AS disbursement_count_180d,

    COALESCE(dh.prior_loan_count_180d, 0)
        AS prior_loan_count_180d,

    COALESCE(dh.prior_active_loan_months_180d, 0)
        AS prior_active_loan_months_180d,

    dh.avg_loan_amount_30d,
    dh.avg_loan_amount_90d,
    dh.avg_loan_amount_180d,

    COALESCE(dh.disbursed_ugx_30d, 0)
        AS disbursed_ugx_30d,

    COALESCE(dh.disbursed_ugx_90d, 0)
        AS disbursed_ugx_90d,

    COALESCE(dh.disbursed_ugx_180d, 0)
        AS disbursed_ugx_180d,

    dh.months_since_first_observed_loan,

    -- Late-payment penalty history (see penalty_history_features above --
    -- sourced from analytics.momo_loan_book_tracker_xtrafloat_loan_state_
    -- daily's penalty ledger, mapped to prior_late_fee_1_count/
    -- prior_late_fee_2_count/prior_total_late_fee_owed_ugx at training-time
    -- naming via SNAPSHOT_TO_TRAINING_COLUMN_MAP in
    -- pd_model/preprocessing/loan_history_features.py)
    COALESCE(ph.late_fee_1_count, 0)
        AS late_fee_1_count,

    COALESCE(ph.late_fee_2_count, 0)
        AS late_fee_2_count,

    COALESCE(ph.total_late_fee_owed_ugx, 0)
        AS total_late_fee_owed_ugx

FROM base b

LEFT JOIN latest_state_features lsf
  ON lsf.msisdn = b.msisdn

LEFT JOIN loan_history_features h
  ON h.msisdn = b.msisdn

LEFT JOIN closed_loan_features cl
  ON cl.msisdn = b.msisdn

LEFT JOIN last_closed_loan_features lc
  ON lc.msisdn = b.msisdn

LEFT JOIN on_time_streak ots
  ON ots.msisdn = b.msisdn

LEFT JOIN disbursement_history_features dh
  ON dh.msisdn = b.msisdn

LEFT JOIN penalty_history_features ph
  ON ph.msisdn = b.msisdn;
